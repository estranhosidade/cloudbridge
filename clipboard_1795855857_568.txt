// ==UserScript==
// @name         ⚡ IMDb
// @namespace    http://tampermonkey.net/
// @version      1.12
// @description  Replaces IMDb title, person, full-credits, episodes and quotes pages with a compact 2002-style frontend.

// @run-at       document-start
// @grant        GM_xmlhttpRequest

// ============================================================================
// MAINTAINER / LLM HANDOFF GUIDE
// ============================================================================
// Read this before changing routes, extraction, navigation, caching, or markup.
// This project is a single-file browser userscript; there is no build step and
// no external runtime dependency. Keep it that way unless there is a concrete
// reason to introduce a dependency.
//
// PRODUCT CONTRACT
//   - Canonical `/title/<tt-id>/` and `/name/<nm-id>/` routes use the existing
//     title and profession-agnostic person models.
//   - `/title/<tt-id>/fullcredits/` and `/name/<nm-id>/fullcredits/` use the
//     auxiliary listing model and render credit departments as grouped rows.
//   - `/title/<tt-id>/episodes/` uses the auxiliary listing model; `?season=`
//     and other query filters are part of the cache key and are preserved.
//   - `/title/<tt-id>/quotes/` and `/name/<nm-id>/quotes/` render extracted
//     quote blocks with available speaker/source attribution.
//   - Only these explicit route families are supported. Keep unrelated IMDb
//     sub-pages native; never turn `/title/` or `/name/` into a blanket match.
//   - Person pages are profession-agnostic: actor, director, writer, producer,
//     cinematographer, editor, composer, crew, and mixed-career profiles all use
//     the same `type: 'name'` model. Do not add one renderer per profession.
//
// FIRST-PAINT / FLASH PREVENTION
//   - installEarlyPageShield() runs immediately after the route allow-list
//     passes, before waiting for <body> or doing any extraction. It sets a
//     theme-matched document background and hides original body children.
//   - Once body.tm-imdb-clean-mode is set, the main stylesheet continues hiding
//     every direct body child except ROOT_ID. Do not reduce this to #__next only:
//     IMDb can mount portals/sibling roots that otherwise flash on release.
//   - releaseEarlyPageShield() is called only after custom markup is visible.
//     Every native-page fallback must release the shield and remove its style.
//   - Keep this shield narrow: unsupported IMDb routes must never be hidden.
//
// OVERVIEW PAGE NAVIGATION
//   - On title pages, Full cast & crew, Episodes and Quotes belong directly
//     below Genres inside `.tm-credit-line`, in the same `.tm-credit-row` grid.
//   - On person pages (including directors), All credits, Full credits and
//     Quotes belong inside `.tm-credit-line` as the final `Pages` row, using the
//     same inline `.tm-title-section-links` treatment and separators. Do not
//     move either set back to a pill/button navigation bar or above the line.
//
// ROUTING / NAVIGATION FLOW
//   1. classifyImdbPath() is the single route classifier; refreshRouteState()
//      publishes `pageKind`, `isTitlePage`, `isNamePage`, and auxiliary flags.
//   2. isSupportedPagePath() defines which internal links may be intercepted.
//   3. shouldRun() decides whether the userscript takes over the current page.
//   4. navigateTo() handles same-document links; navigateViaFetch() obtains
//      destination HTML and parses it with DOMParser. Do not parse the old live
//      document after pushState: it still belongs to the previous route.
//   5. popstate uses the session cache when possible and fetches on cache miss.
//      Keep the route state updated before cache validation/rendering.
//
// DATA EXTRACTION ORDER
//   - Title: parseTitleData(__NEXT_DATA__) -> DOM enhancement/fallback.
//   - Person: parseNameData(__NEXT_DATA__) -> enhanceNameModelFromDom().
//     If Next.js JSON is absent, parseNameDataFromDom() supplies a minimal
//     identity model, then the same DOM enhancement runs.
//   - parseNameStructuredData() reads JSON-LD Person data as a secondary source.
//   - DOM extraction is a compatibility layer, not a replacement for structured
//     payloads. IMDb changes selectors frequently; keep selector fallbacks
//     narrow and verify that they are scoped to the person page/section.
//   - Name extraction must be tolerant of missing fields. Never assume every
//     person has a biography, photo, known-for panel, or every credit category.
//
// PERSON MODEL CONTRACT
//   parseNameData(), parseNameDataFromDom(), and enhanceNameModelFromDom() must
//   return/maintain this shape (empty strings/arrays are valid fallbacks):
//     {
//       type: 'name', id: 'nm…', name: '…', professions: string[],
//       photo: string, photoAlt: string, biography: string, biographyBy: string,
//       knownFor: [{ title, year, href, note, image, subtitle }],
//       filmography: [{ heading, items: [{ title, year, href, note, image,
//                                        subtitle }] }],
//       details: [{ label, value }], allCreditsHref: '/name/nm…/fullcredits/'
//     }
//   Normalize IMDb's varying wrappers (edges/nodes/items/node) before rendering.
//   Keep links as same-origin paths where possible; render strings via
//   htmlEscape() and images via imgAttr(). Never inject raw IMDb HTML into the
//   custom page.
//
// CREDIT EXTRACTION RULES
//   - normalizeNameCreditItem() is the one canonical conversion from an IMDb
//     credit/title object to a display item. Extend it when IMDb changes its
//     payload shape rather than duplicating item parsing elsewhere.
//   - normalizeNameFilmography() converts section/category wrappers.
//   - collectNameFilmographyDeep() is a bounded fallback for unknown nested
//     payloads. Its depth/skip rules intentionally avoid known-for, related,
//     recommendation, navigation, and search arrays. Review false positives
//     before relaxing those rules; arbitrary arrays can contain title cards.
//   - extractNameSectionsFromDom() is a last-resort DOM reader. Heading text
//     determines the department; getNameSectionRoot() must choose the nearest
//     ancestor containing title links to avoid accidentally absorbing the
//     entire page into every department.
//   - mergeNameFilmographySections() combines JSON, JSON-LD and DOM categories.
//     Deduplication is per normalized heading and per title/year/role note so
//     the same production may legitimately appear in more than one department.
//
// RENDERING / SECURITY CONTRACT
//   - renderNamePage() is shared by all person professions. Keep it presentational:
//     extraction/normalization belongs before rendering.
//   - Escape every interpolated text, attribute, URL, and data-derived label.
//     Use htmlEscape(); do not concatenate untrusted IMDb strings as HTML.
//   - Keep person-specific CSS scoped under `#${ROOT_ID}` and preferably under
//     `.tm-name-*` so title pages are not affected.
//   - Do not add an event handler to data-derived HTML. Existing fixed inline
//     handlers are legacy UI behavior; new interactions should use the shared
//     post-render enhancement/event-delegation path.
//
// CACHE / SPA INVARIANTS
//   - Cache keys include the pathname AND query string. This is essential for
//     season-filtered `/episodes/?season=…` pages. Cached models carry an exact
//     route `pageType` (e.g. `title-episodes`), preventing cross-route reuse.
//   - Model changes should remain serializable (plain objects, strings, arrays).
//     Do not put DOM nodes, Maps, Sets, WeakMaps, or functions into cached models.
//   - Async destination fetches use navigateFetchToken to discard stale results.
//     Preserve this guard when adding requests or asynchronous extraction.
//   - A person render must not start trailer search; trailer state is title-only.
//
// SAFE UPGRADE WORKFLOW
//   1. Inspect live IMDb HTML and __NEXT_DATA__ for at least one actor and one
//      non-actor (director/writer/producer) before changing extraction paths.
//   2. Compare JSON, JSON-LD, and DOM results; do not assume one source is complete.
//   3. Preserve the model contract above and merge missing data instead of
//      replacing populated sections with an incomplete fallback.
//   4. Run `node --check imdb.uc.js` after every edit.
//   5. Exercise direct loads, title->person links, person->title links, all four
//      auxiliary route families, season query changes, browser Back/Forward,
//      warm/cold cache, missing JSON, absent biography/photo, and empty quotes.
//      Test in Firefox/Zen as the primary target.
//   6. Report browser tests separately from static checks. A passing syntax check
//      is not proof that IMDb's live selectors or payloads still work.
//
// CURRENT LIMITATIONS TO KEEP VISIBLE
//   - Auxiliary pages are DOM-first because IMDb blocks/varies some page JSON
//     payloads; if a page has no recognized rows, the original IMDb page is
//     restored instead of displaying a misleading empty custom page.
//   - Full credits and episodes extraction reflects rows present in the loaded
//     document. It does not bypass IMDb pagination or load hidden/next pages.
//   - Quotes are best-effort: only quote blocks actually present in the loaded
//     document can be shown; missing quotes remain missing, not inferred.
//   - DOM selectors and payload names are undocumented IMDb internals; inspect
//     live markup before adding speculative selectors when extraction regresses.
//
// ============================================================================
//

    // Keep this allow-list deliberately narrow. IMDb has many title/name
    // sub-pages whose data model is unrelated to the pages handled here.
    function classifyImdbPath(pathname = location.pathname) {
        const clean = String(pathname || '').replace(/\/{2,}/g, '/');
        let match;

        if ((match = clean.match(/^\/title\/(tt\d+)\/fullcredits\/?$/i))) return 'title-fullcredits';
        if ((match = clean.match(/^\/name\/(nm\d+)\/fullcredits\/?$/i))) return 'name-fullcredits';
        if ((match = clean.match(/^\/title\/(tt\d+)\/episodes\/?$/i))) return 'title-episodes';
        if ((match = clean.match(/^\/title\/(tt\d+)\/quotes\/?$/i))) return 'title-quotes';
        if ((match = clean.match(/^\/name\/(nm\d+)\/quotes\/?$/i))) return 'name-quotes';
        if ((match = clean.match(/^\/title\/(tt\d+)\/?$/i))) return 'title';
        if ((match = clean.match(/^\/name\/(nm\d+)\/?$/i))) return 'name';
        return '';
    }

    function refreshRouteState(url = location.href) {
        try {
            const u = new URL(url, location.href);
            path = u.pathname || location.pathname;
        } catch (err) {
            path = location.pathname;
        }

        pageKind = classifyImdbPath(path);
        isTitlePage = pageKind === 'title' || pageKind.startsWith('title-');
        isNamePage = pageKind === 'name' || pageKind.startsWith('name-');
        isAuxiliaryPage = !!pageKind && pageKind !== 'title' && pageKind !== 'name';

        return { path, pageKind, isTitlePage, isNamePage, isAuxiliaryPage };
    }

    refreshRouteState();

    function getNameIdFromUrl(url = location.href) {
        try {
            const parsed = new URL(url, location.href);
            return parsed.pathname.match(/\/name\/(nm\d+)(?:\/|$)/i)?.[1] || '';
        } catch (err) {
            return String(url || '').match(/\/name\/(nm\d+)(?:\/|$)/i)?.[1] || '';
        }
    }

    function getTitleIdFromUrl(url = location.href) {
        try {
            const parsed = new URL(url, location.href);
            return parsed.pathname.match(/\/title\/(tt\d+)(?:\/|$)/i)?.[1] || '';
        } catch (err) {
            return String(url || '').match(/\/title\/(tt\d+)(?:\/|$)/i)?.[1] || '';
        }
    }

    function isSupportedPagePath(pathname) {

    function waitForBody(callback) {
        if (document.body) {
            callback();
            return;
        }

        const observer = new MutationObserver(() => {
            if (!document.body) {
                return;
            }

            observer.disconnect();

            try {
                callback();
            } catch (err) {
                console.error(
                    '[IMDb-2002] Error during body-ready callback:',
                    err
                );
            }
        });

        const observeTarget = document.documentElement || document;

        observer.observe(observeTarget, {
            childList: true,
            subtree: true
        });
    }

    function getNavigationType() {
        try {
            const navEntry =
                performance?.getEntriesByType?.('navigation')?.[0];

                return navEntry.type;

        } catch (err) {}

    function cacheKeyForUrl(url = location.href) {
        const cleanUrl = String(url || '').split('#')[0];

        try {
            const u = new URL(cleanUrl, location.href);
            // Season/page/filter query values change the data returned by IMDb.
            // Sort keys for stable keys without dropping any value.
            u.searchParams.sort();
            return `${SESSION_CACHE_PREFIX}${u.origin}${u.pathname}${u.search}`;
        } catch (err) {
            return `${SESSION_CACHE_PREFIX}${cleanUrl}`;
        }
    }

    function scrollKeyForUrl(url = location.href) {
        const cleanUrl = String(url || '').split('#')[0];

        try {
            const u = new URL(cleanUrl, location.href);
            return `${SESSION_SCROLL_PREFIX}${u.origin}${u.pathname}`;
        } catch (err) {
            return `${SESSION_SCROLL_PREFIX}${cleanUrl}`;
        }
    }

    function readSessionJson(key) {
        try {
            const raw = sessionStorage.getItem(key);

            if (!raw) {
                return null;
            }

            return tryJSON(raw);
        } catch (err) {
            return null;
        }
    }

    function writeSessionJson(key, value) {
        try {
            sessionStorage.setItem(key, JSON.stringify(value));
            return true;
        } catch (err) {
            return false;
        }
    }

    function pruneSessionCache(index) {
        const now = Date.now();

        const nextIndex = safeArray(index)
            .filter(
                entry =>
                    entry &&
                    entry.key &&
                    entry.savedAt &&
                    now - entry.savedAt <= SESSION_CACHE_TTL_MS
            )
            .sort(
                (a, b) =>
                    (b.savedAt || 0) -
                    (a.savedAt || 0)
            )
            .slice(0, SESSION_CACHE_MAX_ENTRIES);

        const keep = new Set(
            nextIndex.map(entry => entry.key)
        );

        const original = safeArray(index);

        for (const entry of original) {
            if (!entry?.key) {
                continue;
            }

            if (!keep.has(entry.key)) {
                try {
                    sessionStorage.removeItem(entry.key);
                } catch (err) {}
            }
        }

        writeSessionJson(
            SESSION_CACHE_INDEX_KEY,
            nextIndex
        );

        return nextIndex;
    }

    function persistPageCache(model) {
        if (!model) {
            return;
        }

        const payload = {
            savedAt: Date.now(),
            url: String(location.href || '').split('#')[0],
            pageType: pageKind,
            trailerVideoId: trailerVideoId || null,
            model
        };

        const key = cacheKeyForUrl();

        if (!writeSessionJson(key, payload)) {
            return;
        }

        const index =
            readSessionJson(SESSION_CACHE_INDEX_KEY) || [];

        const nextIndex = [
            ...safeArray(index).filter(
                entry =>
                    entry?.key &&
                    entry.key !== key
            ),
            {
                key,
                savedAt: payload.savedAt
            }
        ];

        pruneSessionCache(nextIndex);
    }

    function updateCachedTrailer(videoId) {
        if (!videoId) {
            return;
        }

        const key = cacheKeyForUrl();
        const payload = readSessionJson(key);

        if (!payload?.model) {
            return;
        }

        payload.trailerVideoId = videoId;
        payload.savedAt = Date.now();

        writeSessionJson(key, payload);
    }

    function persistScrollPosition() {

        } catch (err) {}
    }

    function loadScrollPosition() {
        try {
            const raw = sessionStorage.getItem(
                scrollKeyForUrl()

            const val = Number(raw);

    function text(value) {
        if (value == null) {
            return '';
        }

        if (typeof value === 'string') {
            return value.trim();
        }

        if (
            typeof value === 'number' ||
            typeof value === 'boolean'
        ) {
            return String(value);
        }

        if (Array.isArray(value)) {
            return value
                .map(text)
                .filter(Boolean)
                .join(', ');
        }

        if (typeof value === 'object') {
            const candidates = [
                value.text,
                value.plainText,
                value.originalText,
                value.value,
                value.label,
                value.title,
                value.name
            ];

            for (const candidate of candidates) {
                const out = text(candidate);

                if (out) {
                    return out;
                }
            }
        }

        return '';
    }

    function htmlEscape(str) {
        return String(str ?? '')
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#39;');
    }

    function parseHumanNumber(input) {
        const raw = text(input)
            .replace(/,/g, '')
            .trim();

        if (!raw) {
            return 0;
        }

        const match = raw.match(
            /(\d+(?:\.\d+)?)\s*([kmb])?/i
        );

        if (!match) {
            return 0;
        }

        const base = Number(match[1]);

        if (!Number.isFinite(base)) {
            return 0;
        }

        const suffix = (
            match[2] || ''
        ).toLowerCase();

        const multiplier =
            suffix === 'k'
                ? 1e3
                : suffix === 'm'
                    ? 1e6
                    : suffix === 'b'
                        ? 1e9
                        : 1;

        return Math.round(
            base * multiplier
        );
    }

    function findObjectsDeep(
        root,
        predicate,
        results = [],
        seen = new WeakSet()
    ) {
        if (!root || typeof root !== 'object') {
            return results;
        }

        if (seen.has(root)) {
            return results;
        }

        seen.add(root);

        if (predicate(root)) {
            results.push(root);
        }

        if (Array.isArray(root)) {
            for (const item of root) {
                findObjectsDeep(
                    item,
                    predicate,
                    results,
                    seen
                );
            }
        } else {
            for (const key of Object.keys(root)) {
                findObjectsDeep(
                    root[key],
                    predicate,
                    results,
                    seen
                );
            }
        }

        return results;
    }

    function findFirstDeep(root, predicate) {
        const found = findObjectsDeep(
            root,
            predicate,
            []
        );

        return found[0] || null;
    }

    function resizeImage(url, width = 400) {

            url.includes('imdb.com')
        ) {

    function imgAttr(
        url,
        width = 400,
        alt = ''
    ) {
        if (!url) {
            return '';
        }

        const optimized =
            resizeImage(url, width);

        const small =
            resizeImage(
                url,
                Math.floor(width / 2)
            );

        const original =
            url.replace(
                /@\._V1_.*\.jpg$/,
                '@._V1_.jpg'
            );

        return `
            src="${htmlEscape(optimized)}"
            srcset="${htmlEscape(small)} ${Math.floor(width / 2)}w, ${htmlEscape(optimized)} ${width}w, ${htmlEscape(original)} 1000w"
            sizes="(max-width: ${width}px) 100vw, ${width}px"
            alt="${htmlEscape(alt)}"
        `;
    }

    function normalizeHref(href) {
        const value = text(href);

        if (!value) {
            return '';
        }

        if (/^https?:\/\//i.test(value)) {
            try {
                const url = new URL(value);

                if (
                    /imdb\.com$/i.test(
                        url.hostname
                    )
                ) {
                    return (
                        `${url.pathname}` +
                        `${url.search}` +
                        `${url.hash}`
                    );
                }
            } catch (error) {
                return value;
            }
        }

        return value;
    }

    function uniqueBy(items, keyFn) {
        const seen = new Set();
        const result = [];

        for (const item of safeArray(items)) {
            if (!item) {
                continue;
            }

            const key = keyFn(item);

            if (!key || seen.has(key)) {
                continue;
            }

            seen.add(key);
            result.push(item);
        }

        return result;
    }

    function mergePeople(
        primary,
        secondary,
        limit = 24
    ) {
        const all = [
            ...safeArray(primary),
            ...safeArray(secondary)
        ].filter(item => item?.text);

        all.sort(
            (a, b) =>
                (b.image ? 1 : 0) -
                (a.image ? 1 : 0)
        );

        return uniqueBy(
            all,
            item =>
                `${normalizeHref(item.href)}|${text(item.text).toLowerCase()}`
        ).slice(0, limit);
    }

    function normalizeSpace(value) {
        return text(value)
            .replace(/\s+/g, ' ')
            .trim();
    }

    function normalizeMultiline(value) {
        const raw = String(
            value || ''
        ).replace(/\r\n?/g, '\n');

        const lines = raw
            .split('\n')
            .map(line =>
                line
                    .replace(/[\t ]+/g, ' ')
                    .trimEnd()
            );

        while (
            lines.length &&
            !lines[0].trim()
        ) {
            lines.shift();
        }

        while (
            lines.length &&
            !lines[lines.length - 1].trim()
        ) {
            lines.pop();
        }

        const out = [];
        let blankRun = 0;

        for (const line of lines) {
            const isBlank = !line.trim();

            if (isBlank) {
                blankRun += 1;

                if (blankRun <= 2) {
                    out.push('');
                }

                continue;
            }

            blankRun = 0;
            out.push(line);
        }

        return out.join('\n').trim();
    }

    function pickLongestText(
        values,
        minLength = 1
    ) {
        const candidates =
            uniqueNonEmptyTexts(values)
                .filter(
                    value =>
                        value.length >= minLength
                )
                .sort(
                    (a, b) =>
                        b.length - a.length
                );

        return candidates[0] || '';
    }

        return candidates[0] || '';
    }

    function preferLongerText(
        primary,
        secondary
    ) {
        const first =
            normalizeSpace(primary);

        const second =
            normalizeSpace(secondary);

        if (!first) return second;
        if (!second) return first;

        return second.length > first.length
            ? second
            : first;
    }

        const first =
            normalizeMultiline(primary);

        const second =
            normalizeMultiline(secondary);

        if (!first) return second;
        if (!second) return first;

        return second.length > first.length
            ? second
            : first;
    }

    // Canonical credit adapter. IMDb wraps title data differently across
    // __NEXT_DATA__, JSON-LD, and DOM-derived objects; all downstream credit
    // sections should consume this normalized display-item shape.
    function normalizeNameCreditItem(entry) {
        if (!entry || typeof entry !== 'object') {
            return null;
        }

        let node =
            entry.node ||
            entry.credit?.node ||
            (entry.title && typeof entry.title === 'object'
                ? (entry.title.node || entry.title)
                : entry);

        const nestedTitle =
            node?.title && typeof node.title === 'object'
                ? (node.title.node || node.title)
                : node?.titleSummary ||
                  node?.titleData ||
                  node;

        const title = text(
            node?.titleText ||
            nestedTitle?.titleText ||
            node?.originalTitleText ||
            nestedTitle?.originalTitleText ||
            node?.primaryTitle ||
            nestedTitle?.primaryTitle ||
            node?.nameText ||
            nestedTitle?.nameText ||
            (typeof node?.title === 'string' ? node.title : '') ||
            (typeof nestedTitle === 'string' ? nestedTitle : '') ||
            node?.name ||
            nestedTitle?.name
        );

        if (!title) {
            return null;
        }

        const candidateId =
            text(node?.id) ||
            text(node?.['@id']) ||
            text(nestedTitle?.id) ||
            text(nestedTitle?.['@id']) ||
            text(node?.titleId) ||
            text(entry?.titleId);
        const titleId = candidateId.match(/tt\d+/i)?.[0] || '';
        const rawHref =
            node?.href ||
            nestedTitle?.href ||
            node?.url ||
            node?.['@id'] ||
            nestedTitle?.url ||
            nestedTitle?.['@id'] ||
            entry?.href ||
            '';

        let href = titleId
            ? `/title/${titleId}/`
            : normalizeHref(rawHref);

        if (href && !/^https?:\/\//i.test(href) && !href.startsWith('/')) {
            href = `/${href}`;
        }

        const hasTitleIdentity =
            !!titleId ||
            /\/title\/tt\d+/i.test(href) ||
            !!node?.titleType ||
            !!node?.releaseYear ||
            !!node?.primaryImage;

        if (!hasTitleIdentity) {
            return null;
        }

        const year =
            text(node?.releaseYear?.year) ||
            text(nestedTitle?.releaseYear?.year) ||
            text(node?.year) ||
            text(nestedTitle?.year) ||
            text(node?.startYear?.year) ||
            text(node?.startYear) ||
            text(node?.datePublished) ||
            text(nestedTitle?.datePublished) ||
            '';

        const note = text(
            entry?.attributes ||
            entry?.attribute ||
            entry?.characters ||
            entry?.character ||
            entry?.episodeCredits ||
            entry?.role ||
            entry?.job ||
            entry?.notes ||
            entry?.summary ||
            node?.attributes ||
            node?.characters ||
            node?.episodeCredits ||
            node?.role ||
            node?.job ||
            node?.summary
        );

        const image =
            node?.primaryImage?.url ||
            nestedTitle?.primaryImage?.url ||
            node?.image?.url ||
            nestedTitle?.image?.url ||
            (typeof node?.image === 'string' ? node.image : '') ||
            (typeof nestedTitle?.image === 'string' ? nestedTitle.image : '') ||
            '';

        return {
            title,
            year,
            href,
            note,
            image,
            subtitle: text(node?.titleType?.text || node?.titleType || node?.category || '')
        };
    }

    function nameCreditSectionHeading(section, fallback = 'Credits') {
        const explicit = text(
            section?.heading ||
            section?.category ||
            section?.displayName ||
            section?.jobCategory?.text ||
            section?.jobCategory ||
            section?.title ||
            section?.section ||
            section?.label ||
            section?.name ||
            section?.job ||
            section?.department
        );

        if (explicit) {
            return explicit;
        }

        return String(fallback || 'Credits')
            .replace(/([a-z])([A-Z])/g, '$1 $2')
            .replace(/[_-]+/g, ' ')
            .replace(/\b\w/g, c => c.toUpperCase());
    }

    // Convert category/department containers (edges, nodes, items, jobs, etc.)
    // into [{ heading, items }]. Keep wrapper handling here instead of adding
    // format-specific branches to renderNamePage().
    function normalizeNameFilmography(source, fallbackHeading = 'Credits') {
        if (!source) {
            return [];
        }

        const asList = value => {
            if (Array.isArray(value)) return value;
            return safeArray(value?.edges || value?.items || value?.nodes);
        };

        const rawSectionSource = Array.isArray(source)
            ? source
            : source.categories ||
              source.sections ||
              source.items ||
              source.jobs ||
              source.creditsByCategory;
        let rawSections = asList(rawSectionSource);

        const sourceHasDirectCredits =
            !rawSections.length &&
            asList(source.edges || source.credits || source.titles || source.nodes).length;

        if (Array.isArray(source)) {
            const directItems = source.map(normalizeNameCreditItem).filter(Boolean);
            if (directItems.length) {
                rawSections = [{ category: fallbackHeading, credits: source }];
            }
        }

        const sections = rawSections.length
            ? rawSections
            : sourceHasDirectCredits
                ? [source]
                : [];

        return sections
            .map((section, index) => {
                const rawItems =
                    section?.credits ||
                    section?.items ||
                    section?.edges ||
                    section?.titles ||
                    section?.nodes ||
                    section?.entries ||
                    section?.jobs ||
                    [];
                const itemList = Array.isArray(rawItems)
                    ? rawItems
                    : safeArray(rawItems?.edges || rawItems?.items || rawItems?.nodes);

                const items = itemList
                    .map(normalizeNameCreditItem)
                    .filter(Boolean);

                return {
                    heading: nameCreditSectionHeading(
                        section,
                        `${fallbackHeading} ${index + 1}`
                    ),
                    items: uniqueBy(
                        items,
                        item => `${item.href || item.title}|${item.year}|${item.note}`
                    ).slice(0, 100)
                };
            })
            .filter(section => section.items.length);
    }

    /**
     * Merge all available person-credit sources without allowing a partially
     * populated JSON payload to hide categories discovered in the DOM/JSON-LD.
     *
     * `sectionsBySource` are ordered by precedence: earlier sources establish
     * section order and display spelling; later sources fill missing entries.
     * Items are deduplicated within a department by title URL, year, and role
     * note. The note is part of identity because one title can have multiple
     * credited characters/jobs. A title may also appear in different departments.
     */
    function mergeNameFilmographySections(...sectionsBySource) {
        const merged = new Map();

        for (const source of sectionsBySource) {
            for (const section of safeArray(source)) {
                const heading = nameCreditSectionHeading(section, 'Credits');
                const headingKey = normalizeSpace(heading)
                    .toLocaleLowerCase()
                    .replace(/\s+/g, ' ');
                if (!headingKey) continue;

                if (!merged.has(headingKey)) {
                    merged.set(headingKey, {
                        heading,
                        items: []
                    });
                }

                const target = merged.get(headingKey);
                target.items.push(...safeArray(section?.items));
            }
        }

        return [...merged.values()]
            .map(section => ({
                heading: section.heading,
                items: uniqueBy(
                    section.items,
                    item => {
                        const href = normalizeHref(item?.href) || text(item?.title);
                        return `${href}|${text(item?.year)}|${text(item?.note)}`;
                    }
                ).slice(0, 100)
            }))
            .filter(section => section.items.length)
            .slice(0, 40);
    }

    function collectNameFilmographyDeep(root) {
        const sections = [];
        const seen = new WeakSet();
        const visitedArrays = new WeakSet();
        const skipPath = /known.?for|recommend|related|suggest|navigation|search|trending|popular/i;

        function visit(value, path = [], depth = 0) {
            if (!value || typeof value !== 'object' || depth > 14 || seen.has(value)) {
                return;
            }
            seen.add(value);

            if (Array.isArray(value)) {
                if (!visitedArrays.has(value)) {
                    visitedArrays.add(value);
                    const pathLabel = path.join(' ');
                    if (!skipPath.test(pathLabel)) {
                        const items = value
                            .map(normalizeNameCreditItem)
                            .filter(Boolean);
                        if (items.length >= 2) {
                            const heading = path
                                .filter(part => !/^\d+$/.test(part) && !/^(edges|nodes|items|credits|titles|filmography)$/i.test(part))
                                .slice(-2)
                                .join(' ') || 'Credits';
                            sections.push({
                                heading: nameCreditSectionHeading(null, heading),
                                items
                            });
                        }
                    }
                }
                value.forEach((item, index) => visit(item, [...path, String(index)], depth + 1));
                return;
            }

            for (const [key, child] of Object.entries(value)) {
                if (child && typeof child === 'object') {
                    visit(child, [...path, key], depth + 1);
                }
            }
        }

        visit(root);

        const merged = new Map();
        for (const section of sections) {
            const key = text(section.heading).toLowerCase();
            if (!merged.has(key)) {
                merged.set(key, { heading: section.heading, items: [] });
            }
            const target = merged.get(key);
            target.items.push(...section.items);
        }

        return [...merged.values()]
            .map(section => ({
                heading: section.heading,
                items: uniqueBy(
                    section.items,
                    item => `${item.href || item.title}|${item.year}|${item.note}`
                ).slice(0, 100)
            }))
            .filter(section => section.items.length);
    }

    function getNameSectionRoot(heading) {
        if (!heading) {
            return null;
        }

        // Prefer the nearest ancestor that actually contains title links.
        // Starting from the heading (rather than choosing the first <section>)
        // prevents every job heading from inheriting the entire Credits panel.
        let current = heading;
        for (let depth = 0; current && depth < 10; depth++, current = current.parentElement) {
            const count = current.querySelectorAll?.('a[href*="/title/"]').length || 0;
            if (count) {
                return current;
            }
        }

        return null;
    }

    function collectNameTitleItemsFromDom(root, limit = 100) {
        if (!root?.querySelectorAll) {
            return [];
        }

        const items = [];
        const anchors = Array.from(
            root.querySelectorAll('a[href*="/title/"]')
        );

        for (const anchor of anchors) {
            const href = normalizeHref(anchor.getAttribute('href') || anchor.href);
            const titleId = href.match(/\/title\/(tt\d+)/i)?.[1];
            if (!titleId) {
                continue;
            }

            const card = anchor.closest?.(
                '[data-testid*="credit"], [data-testid*="title"], .ipc-metadata-list-summary-item, .ipc-metadata-list-item, li, article'
            ) || anchor.parentElement || anchor;
            const imageEl = card.querySelector?.('img');
            const title =
                text(anchor.textContent) ||
                text(anchor.getAttribute('aria-label')) ||
                text(imageEl?.getAttribute('alt'));
            if (!title) {
                continue;
            }

            const cardText = normalizeSpace(card.textContent || '');
            const yearMatch = cardText.match(/\b((?:18|19|20|21)\d{2})\b/);
            const note = pickDomText(card, [
                '[data-testid*="character"]',
                '[data-testid*="role"]',
                '[class*="character"]',
                '[class*="role"]',
                '[data-testid*="credit-note"]'
            ]);
            const image =
                imageEl?.getAttribute('src') ||
                imageEl?.getAttribute('data-src') ||
                imageEl?.getAttribute('srcset')?.split(',')?.[0]?.trim()?.split(/\s+/)?.[0] ||
                '';

            items.push({
                title,
                year: yearMatch?.[1] || '',
                href: `/title/${titleId}/`,
                note,
                image,
                subtitle: ''
            });
        }

        return uniqueBy(
            items,
            item => `${item.href}|${item.note}|${item.year}`
        ).slice(0, limit);
    }

    function extractNameSectionsFromDom(doc) {
        if (!doc?.querySelectorAll) {
            return { knownFor: [], filmography: [] };
        }

        const cached = NAME_DOM_SECTIONS_CACHE.get(doc);
        if (cached) {
            return cached;
        }

        const headings = Array.from(
            doc.querySelectorAll('h2, h3, [role="heading"], [data-testid*="section-title"]')
        );
        let knownFor = [];
        const filmography = [];
        const seenSections = new Set();

        for (const heading of headings) {
            const label = normalizeSpace(heading.textContent || '');
            if (!label) {
                continue;
            }

            const root = getNameSectionRoot(heading);
            if (!root) {
                continue;
            }

            const items = collectNameTitleItemsFromDom(root, 150);
            if (!items.length) {
                continue;
            }

            if (/^known\s*for\b/i.test(label)) {
                if (!knownFor.length) {
                    knownFor = items.slice(0, 20);
                }
                continue;
            }

            if (/biography|personal details|photos?|videos?|news|awards|connections|trivia|contact|client/i.test(label)) {
                continue;
            }

            if (!/credits?|filmography|acting|actor|actress|directing|director|writing|writer|producing|producer|cinematograph|camera|editing|editor|composer|music|sound|art department|crew|self|archive footage|thanks|miscellaneous/i.test(label)) {
                continue;
            }

            const key = `${label.toLowerCase()}|${items.map(item => item.href).join(',')}`;
            if (seenSections.has(key)) {
                continue;
            }
            seenSections.add(key);

            filmography.push({
                heading: label,
                items: items.slice(0, 100)
            });
        }

        // Some layouts have no useful heading hierarchy but expose a Credits section test id.
        if (!filmography.length) {
            const roots = Array.from(doc.querySelectorAll(
                '[data-testid*="credit" i], [data-testid*="filmography" i], [class*="filmography"]'
            ));
            for (const root of roots.slice(0, 12)) {
                const items = collectNameTitleItemsFromDom(root, 150);
                if (items.length) {
                    const heading = pickDomText(root, ['h2', 'h3', '[role="heading"]']) || 'Credits';
                    filmography.push({ heading, items: items.slice(0, 100) });
                }
            }
        }

        const result = {
            knownFor,
            filmography: filmography.slice(0, 40)
        };

        if (result.knownFor.length || result.filmography.length) {
            NAME_DOM_SECTIONS_CACHE.set(doc, result);
        }

        return result;
    }

    function extractNameDetailsFromDom(doc) {
        if (!doc?.querySelectorAll) {
            return [];
        }

        const details = [];
        const push = (label, value) => {
            label = text(label).replace(/\s+/g, ' ').replace(/:$/, '');
            value = text(value);
            if (label && value && !details.some(item => item.label.toLowerCase() === label.toLowerCase())) {
                details.push({ label, value });
            }
        };

        const detailRoots = Array.from(doc.querySelectorAll(
            '[data-testid*="personal-details" i], [data-testid*="bio-content" i], .ipc-metadata-list'
        ));

        for (const root of detailRoots.slice(0, 12)) {
            for (const item of Array.from(root.querySelectorAll('li, tr, .ipc-metadata-list__item')).slice(0, 50)) {
                const label = pickDomText(item, [
                    'dt', 'th', '[data-testid*="label"]', '.ipc-metadata-list-item__label'
                ]);
                const value = pickDomText(item, [
                    'dd', 'td', '[data-testid*="value"]', '.ipc-metadata-list-item__content'
                ]) || normalizeSpace(item.textContent || '').replace(label, '').trim();
                push(label, value);
            }
        }

        return details;
    }

    // Parse the Name page's Next.js payload. This function must remain
    // profession-neutral: the same model feeds actors, directors, writers,
    // producers, and other crew profiles. See the maintainer model contract at
    // the top of this file before adding/renaming model fields.
    function parseNameData(data, sourceUrl = location.href) {
        const pageProps =
            data?.props?.pageProps ||
            data?.pageProps ||
            {};
        const expectedId = getNameIdFromUrl(sourceUrl);

        const maybeNameRoots = findObjectsDeep(
            pageProps,
            obj => {
                const id = typeof obj?.id === 'string' && /^nm\d+/i.test(obj.id);
                const hasName = !!text(obj?.nameText || obj?.displayName || obj?.titleText || obj?.name);
                const hasBio = !!(
                    obj?.miniBios || obj?.miniBio || obj?.bio ||
                    obj?.biography || obj?.knownFor || obj?.filmography ||
                    obj?.primaryProfessions || obj?.professions || obj?.credits
                );
                return id || (hasName && hasBio);
            }
        );

        const scorePerson = obj => {
            let score = 0;
            if (obj?.id && expectedId && obj.id === expectedId) score += 1000;
            if (obj?.id && /^nm\d+/i.test(String(obj.id))) score += 40;
            if (text(obj?.nameText || obj?.displayName || obj?.titleText || obj?.name)) score += 25;
            if (obj?.primaryImage?.url || obj?.heroImage?.url) score += 12;
            if (obj?.primaryProfessions || obj?.professions || obj?.jobCategories) score += 12;
            if (obj?.miniBios || obj?.miniBio || obj?.biography || obj?.bio) score += 12;
            if (obj?.knownFor || obj?.knownForTitles || obj?.knownForCredits) score += 10;
            if (obj?.filmography || obj?.credits || obj?.creditsByCategory) score += 10;
            if (obj?.personalDetails || obj?.bioDetails) score += 6;
            return score;
        };

        const person =
            (expectedId
                ? maybeNameRoots.find(obj => obj?.id === expectedId)
                : null) ||
            maybeNameRoots
                .filter(obj => !obj?.id || !/^nm\d+/i.test(String(obj.id)))
                .sort((a, b) => scorePerson(b) - scorePerson(a))[0] ||
            pageProps.aboveTheFoldData ||
            pageProps.mainColumnData ||
            pageProps.person ||
            {};

        const knownForSource =
            person.knownFor ||
            person.knownForTitles ||
            person.knownForCredits ||
            person.knownForMovies ||
            findFirstDeep(
                person,
                obj =>
                    Array.isArray(obj?.edges) &&
                    obj.edges.some(edge => /^tt\d+/i.test(edge?.node?.id || edge?.id || ''))
            );

        const knownForEdges = safeArray(
            knownForSource?.edges ||
            knownForSource?.titles ||
            knownForSource?.items ||
            knownForSource?.nodes ||
            knownForSource
        );

        const filmographySource =
            person.filmography ||
            person.creditsByCategory ||
            person.primaryCredits ||
            person.credits ||
            person.jobCategories ||
            findFirstDeep(
                person,
                obj => obj && typeof obj === 'object' && (
                    Array.isArray(obj?.categories) ||
                    Array.isArray(obj?.categories?.edges) ||
                    Array.isArray(obj?.sections) ||
                    Array.isArray(obj?.sections?.edges)
                )
            );

        let filmography = normalizeNameFilmography(
            filmographySource,
            'Credits'
        );

        if (!filmography.length) {
            filmography = collectNameFilmographyDeep(person);
        }

        const detailsSource =
            person.personalDetails ||
            person.bioDetails ||
            findFirstDeep(
                person,
                obj => obj && typeof obj === 'object' && (
                    obj.birthDate || obj.deathDate || obj.birthLocation ||
                    obj.birthPlace || obj.height || obj.nicknames
                )
            ) || {};

        const miniBioSource = safeArray(
            person.miniBios ||
            person.miniBio ||
            person.biographies ||
            []
        );
        const mainBio = miniBioSource[0] || person.biography || person.bio || {};
        const birthDate = detailsSource?.birthDate;
        const deathDate = detailsSource?.deathDate;
        const details = [];

        const maybeDetailMap = [
            ['Born', text(birthDate?.date || birthDate?.displayableProperty || birthDate) || text(person.birthDate?.date || person.birthDate)],
            ['Birth place', text(detailsSource?.birthLocation || detailsSource?.birthPlace || person.birthLocation || person.birthPlace)],
            ['Died', text(deathDate?.date || deathDate?.displayableProperty || deathDate) || text(person.deathDate?.date || person.deathDate)],
            ['Death place', text(detailsSource?.deathLocation || detailsSource?.deathPlace || person.deathLocation)],
            ['Height', text(detailsSource?.height || person.height)],
            ['Nicknames', joinItems(detailsSource?.nicknames || detailsSource?.nickname || person.nicknames, '')],
            ['Alternate names', joinItems(detailsSource?.akas || detailsSource?.aka || detailsSource?.alternativeNames || person.akas, '')],
            ['Spouse', text(detailsSource?.spouse || person.spouse)],
            ['Children', text(detailsSource?.children || person.children)]
        ];

        for (const [label, value] of maybeDetailMap) {
            if (value) details.push({ label, value });
        }

        const name = text(
            person.nameText ||
            person.displayName ||
            person.titleText ||
            person.name ||
            pageProps.nameText ||
            pageProps.displayName
        );

        const id =
            expectedId ||
            text(person.id).match(/nm\d+/i)?.[0] ||
            '';

        return {
            type: 'name',
            id,
            name,
            professions: uniqueStrings(
                person.primaryProfessions ||
                person.professions ||
                person.jobCategories ||
                person.profession ||
                []
            ),
            photo:
                person.primaryImage?.url ||
                person.heroImage?.url ||
                person.image?.url ||
                '',
            photoAlt: name,
            biography: text(
                mainBio?.text?.plainText ||
                mainBio?.text?.originalText?.plainText ||
                mainBio?.plainText ||
                mainBio?.text ||
                mainBio?.originalText ||
                person.biography?.plainText ||
                person.biography
            ),
            biographyBy: text(
                mainBio?.author?.displayName ||
                mainBio?.author?.name ||
                mainBio?.author
            ),
            knownFor: uniqueBy(
                knownForEdges.map(normalizeNameCreditItem).filter(Boolean),
                item => item.href || item.title
            ).slice(0, 20),
            filmography,
            details,
            allCreditsHref: id ? `/name/${id}/fullcredits/` : ''
        };
    }

    function parseNameStructuredData(doc, sourceUrl = location.href) {
        if (!doc?.querySelectorAll) {
            return null;
        }

        const cached = NAME_STRUCTURED_DATA_CACHE.get(doc);
        if (cached?.sourceUrl === sourceUrl) {
            return cached.model;
        }

        const expectedId = getNameIdFromUrl(sourceUrl);
        const personObjects = [];

        for (const script of Array.from(
            doc.querySelectorAll('script[type="application/ld+json"]') || []
        )) {
            let payload;
            try {
                payload = JSON.parse(script.textContent || '');
            } catch (err) {
                continue;
            }

            const found = findObjectsDeep(
                payload,
                obj => {
                    const types = safeArray(obj?.['@type']).length
                        ? safeArray(obj['@type'])
                        : [obj?.['@type']];
                    const isPerson = types.some(type => String(type || '').toLowerCase() === 'person');
                    if (!isPerson) return false;

                    const identity = `${obj?.url || ''} ${obj?.['@id'] || ''}`;
                    const hasExpectedId = expectedId && identity.includes(expectedId);
                    return !!hasExpectedId || !!text(obj?.name);
                }
            );

            personObjects.push(...found);
        }

        const person =
            (expectedId
                ? personObjects.find(obj =>
                    `${obj?.url || ''} ${obj?.['@id'] || ''}`.includes(expectedId)
                )
                : null) ||
            personObjects[0];

        if (!person) {
            return null;
        }

        const jobTitles = Array.isArray(person.jobTitle)
            ? person.jobTitle
            : person.jobTitle
                ? [person.jobTitle]
                : [];

        const normalizeStructuredTitles = value => {
            const values = Array.isArray(value) ? value : value ? [value] : [];
            return values.map(item => {
                if (typeof item === 'string') {
                    return { name: item, titleType: 'Title' };
                }
                return item;
            }).map(normalizeNameCreditItem).filter(Boolean);
        };

        const image =
            typeof person.image === 'string'
                ? person.image
                : person.image?.url ||
                  person.image?.contentUrl ||
                  person.image?.thumbnailUrl ||
                  '';

        const birthDate = text(person.birthDate);
        const deathDate = text(person.deathDate);
        const birthPlace = text(person.birthPlace?.name || person.birthPlace);
        const deathPlace = text(person.deathPlace?.name || person.deathPlace);
        const details = [];

        if (birthDate) details.push({ label: 'Born', value: birthDate });
        if (birthPlace) details.push({ label: 'Birth place', value: birthPlace });
        if (deathDate) details.push({ label: 'Died', value: deathDate });
        if (deathPlace) details.push({ label: 'Death place', value: deathPlace });

        const knownFor = normalizeStructuredTitles(person.knownFor);
        const performances = normalizeStructuredTitles(person.performerIn);
        const idFromStructuredUrl = `${person.url || ''} ${person['@id'] || ''}`.match(/nm\d+/i)?.[0] || '';
        const id = expectedId || idFromStructuredUrl;

        const model = {
            type: 'name',
            id,
            name: text(person.name),
            professions: uniqueStrings(jobTitles),
            photo: image,
            photoAlt: text(person.name),
            biography: text(person.description),
            biographyBy: '',
            knownFor: uniqueBy(knownFor, item => item.href || item.title).slice(0, 20),
            filmography: performances.length
                ? [{ heading: 'Performances', items: performances.slice(0, 100) }]
                : [],
            details,
            allCreditsHref: id ? `/name/${id}/fullcredits/` : ''
        };

        NAME_STRUCTURED_DATA_CACHE.set(doc, {
            sourceUrl,
            model
        });

        return model;
    }

    // Overlay secondary data sources onto the normalized person model. Do not
    // treat DOM extraction as an all-or-nothing fallback: IMDb can expose a
    // director section in the DOM while the JSON payload only contains acting
    // credits (or vice versa), so merge each department independently.
    function enhanceNameModelFromDom(model, doc, sourceUrl = location.href) {
        if (!doc || !model) {
            return model;
        }

        const firstImage = selectors => {
            for (const selector of selectors) {
                const img = doc.querySelector(selector);
                if (!img) continue;
                const candidates = [
                    img.currentSrc,
                    img.getAttribute?.('src'),
                    img.getAttribute?.('data-src'),
                    img.getAttribute?.('data-image-lazy-src'),
                    img.getAttribute?.('srcset')?.split(',')?.[0]?.trim()?.split(/\s+/)?.[0]
                ];
                const value = candidates.map(text).find(value => value && !/^data:/i.test(value));
                if (value) return value;
            }
            return '';
        };

        const structured = parseNameStructuredData(doc, sourceUrl) || {};

        model.id = model.id || structured.id || getNameIdFromUrl(sourceUrl);
        model.name = model.name || structured.name || pickDomText(doc, [
            '[data-testid="hero__primary-text"]',
            '[data-testid*="hero__primary-text"]',
            'h1'
        ]);

        model.photo = model.photo || structured.photo || firstImage([
            'img[data-testid="nm_pd_img"]',
            '[data-testid="hero-media__photo"] img',
            '[data-testid="hero-media__poster"] img',
            '[data-testid*="hero-media"] img',
            '[data-testid*="name-hero"] img',
            'main img.ipc-image'
        ]);

        model.biography = model.biography || structured.biography || pickDomHtmlText(doc, [
            '[data-testid="name-bio-text"]',
            '#name-bio-text',
            '[data-testid="biography"]',
            '[data-testid*="biography"] .ipc-html-content-inner-div',
            '[data-testid*="bio"] .ipc-html-content-inner-div'
        ]);

        if (!safeArray(model.professions).length && safeArray(structured.professions).length) {
            model.professions = structured.professions;
        }

        if (!safeArray(model.professions).length) {
            model.professions = uniqueStrings(Array.from(
                doc.querySelectorAll(
                    '[data-testid="hero__metadata"] li, ' +
                    '[data-testid="hero__metadata"] a, ' +
                    '[data-testid*="profession"]'
                ) || []
            ).map(el => el?.textContent));
        }

        const domSections = extractNameSectionsFromDom(doc);
        model.knownFor = uniqueBy(
            [
                ...safeArray(model.knownFor),
                ...safeArray(structured.knownFor),
                ...safeArray(domSections.knownFor)
            ],
            item => item.href || item.title
        ).slice(0, 30);

        model.filmography = mergeNameFilmographySections(
            model.filmography,
            structured.filmography,
            domSections.filmography
        );

        const structuredDetails = safeArray(structured.details);
        const domDetails = extractNameDetailsFromDom(doc);
        model.details = uniqueBy(
            [
                ...safeArray(model.details),
                ...structuredDetails,
                ...domDetails
            ],
            item => text(item?.label).toLowerCase()
        );

        model.photoAlt = model.photoAlt || model.name;
        model.allCreditsHref = model.allCreditsHref || (model.id ? `/name/${model.id}/fullcredits/` : '');
        model.professions = uniqueStrings(model.professions);
        model.knownFor = safeArray(model.knownFor);
        model.filmography = safeArray(model.filmography);
        model.details = safeArray(model.details);

        return model;
    }

    // ---------------------------------------------------------------------
    // Auxiliary IMDb pages: full credits, episodes, and quotes
    // ---------------------------------------------------------------------

    function normalizeAuxiliaryText(value) {
        return String(value || '')
            .replace(/\u00a0/g, ' ')
            .replace(/[\t\r\n]+/g, ' ')
            .replace(/\s{2,}/g, ' ')
            .trim();
    }

    function auxiliaryNodeText(node) {
        if (!node) return '';
        return normalizeAuxiliaryText(node.innerText || node.textContent || '');
    }

    function auxiliarySafeLocalHref(anchor, baseUrl) {
        try {
            const url = new URL(anchor?.getAttribute?.('href') || anchor?.href || '', baseUrl);
            if (url.origin !== new URL(baseUrl, location.href).origin) return '';
            if (!/^\/(?:title\/tt\d+|name\/nm\d+)(?:\/|$)/i.test(url.pathname)) return '';
            return `${url.pathname}${url.search}${url.hash}`;
        } catch (err) {
            return '';
        }
    }

    function auxiliaryImageFromRow(row) {
        const image = row?.querySelector?.('img');
        if (!image) return '';
        return image.currentSrc || image.src || image.getAttribute('data-src') || image.getAttribute('src') || '';
    }

    function auxiliaryRowForAnchor(anchor, scope) {
        const preferred = 'li, tr, article, [data-testid*="summary"], [data-testid*="item"], .ipc-metadata-list-summary-item, .sodatext, .quote';
        let row = anchor.closest?.(preferred);
        if (row && scope?.contains?.(row)) return row;

        row = anchor.parentElement;
        for (let i = 0; row && row !== scope && i < 4; i++, row = row.parentElement) {
            const count = row.querySelectorAll?.('a[href]')?.length || 0;
            if (count <= 4 && auxiliaryNodeText(row).length < 1200) return row;
        }
        return anchor.parentElement || anchor;
    }

    function auxiliaryItemFromAnchor(anchor, scope, baseUrl, kind) {
        const href = auxiliarySafeLocalHref(anchor, baseUrl);
        const label = auxiliaryNodeText(anchor);
        if (!href || !label || label.length > 260) return null;

        const row = auxiliaryRowForAnchor(anchor, scope);
        const links = Array.from(row?.querySelectorAll?.('a[href]') || []);
        const linkedText = links.map(auxiliaryNodeText).filter(Boolean);
        let note = auxiliaryNodeText(row);
        for (const linked of linkedText) {
            if (linked) note = note.replace(linked, ' ');
        }
        note = normalizeAuxiliaryText(note)
            .replace(/^[\s·•|,:–—-]+|[\s·•|,:–—-]+$/g, '');
        if (note === label) note = '';

        const year = (auxiliaryNodeText(row).match(/\b(?:18|19|20|21)\d{2}\b/) || [])[0] || '';
        const isTitleLink = /\/title\/tt\d+/i.test(href);
        const isPersonLink = /\/name\/nm\d+/i.test(href);
        return {
            title: label,
            name: label,
            href,
            note,
            subtitle: note,
            year,
            image: auxiliaryImageFromRow(row),
            kind: isPersonLink ? 'person' : isTitleLink ? 'title' : kind,
            rawText: auxiliaryNodeText(row)
        };
    }

    function auxiliaryHeadingText(node) {
        return auxiliaryNodeText(node)
            .replace(/\s+\(\d+[\d,]*\)\s*$/, '')
            .trim();
    }

    function auxiliarySectionRoot(heading, scope, linkPattern) {
        let current = heading?.parentElement;
        let fallback = null;
        for (let depth = 0; current && current !== scope && depth < 7; depth++, current = current.parentElement) {
            const matches = Array.from(current.querySelectorAll('a[href]')).filter(a => linkPattern.test(a.getAttribute('href') || a.href || ''));
            if (!matches.length) continue;
            fallback = current;
            // A semantic section is useful only when it represents this one
            // heading. A page-wide section containing multiple departments
            // must not be reused for every heading.
            const nestedHeadings = current.querySelectorAll?.('h2,h3,h4,[role="heading"]')?.length || 0;
            if (nestedHeadings <= 1 && current.matches?.('section, article, [data-testid*="section"], [data-testid*="list"]')) return current;
            if (nestedHeadings <= 1) return current;
        }
        return fallback && (fallback.querySelectorAll?.('h2,h3,h4,[role="heading"]')?.length || 0) <= 1 ? fallback : null;
    }

    function extractAuxiliarySections(doc, baseUrl, kind) {
        const isTitleCredits = kind === 'title-fullcredits';
        const pattern = isTitleCredits ? /\/name\/nm\d+/i : /\/title\/tt\d+/i;
        const scope = doc.querySelector('main') || doc.querySelector('#main') || doc.body || doc;
        const headings = Array.from(scope.querySelectorAll('h2,h3,h4,h5,[role="heading"]'));
        const sections = [];
        const usedRoots = new Set();

        for (const heading of headings) {
            const headingText = auxiliaryHeadingText(heading);
            if (!headingText || headingText.length > 100) continue;
            const root = auxiliarySectionRoot(heading, scope, pattern);
            if (!root || usedRoots.has(root)) continue;
            const anchors = Array.from(root.querySelectorAll('a[href]')).filter(a => pattern.test(a.getAttribute('href') || a.href || ''));
            const items = [];
            const seen = new Set();
            for (const anchor of anchors) {
                const item = auxiliaryItemFromAnchor(anchor, root, baseUrl, isTitleCredits ? 'person' : 'title');
                if (!item) continue;
                const key = `${item.href}|${item.title}|${item.year}|${item.note}`.toLowerCase();
                if (seen.has(key)) continue;
                seen.add(key);
                items.push(item);
            }
            if (items.length) {
                usedRoots.add(root);
                sections.push({ heading: headingText, items });
            }
        }

        // Some IMDb variants render one flat credits list with no semantic
        // heading hierarchy. Preserve that list under a neutral heading.
        const covered = new Set(sections.flatMap(section => section.items.map(item => item.href.toLowerCase())));
        const unassigned = [];
        const unassignedSeen = new Set();
        for (const anchor of Array.from(scope.querySelectorAll('a[href]')).filter(a => pattern.test(a.getAttribute('href') || a.href || ''))) {
            const item = auxiliaryItemFromAnchor(anchor, scope, baseUrl, isTitleCredits ? 'person' : 'title');
            if (!item || covered.has(item.href.toLowerCase())) continue;
            const key = `${item.href}|${item.title}|${item.year}|${item.note}`.toLowerCase();
            if (!unassignedSeen.has(key)) { unassignedSeen.add(key); unassigned.push(item); }
        }
        if (unassigned.length) {
            sections.push({ heading: sections.length ? 'Additional credits' : (isTitleCredits ? 'Cast & crew' : 'Credits'), items: unassigned });
        }
        return sections;
    }

    function extractAuxiliaryEpisodes(doc, baseUrl, titleId) {
        const scope = doc.querySelector('main') || doc.querySelector('#main') || doc.body || doc;
        const anchors = Array.from(scope.querySelectorAll('a[href]')).filter(anchor => {
            const href = auxiliarySafeLocalHref(anchor, baseUrl);
            return /\/title\/tt\d+/i.test(href) && !new RegExp(`/title/${titleId}(?:/|$)`, 'i').test(href);
        });
        const items = [];
        const seen = new Set();
        for (const anchor of anchors) {
            const item = auxiliaryItemFromAnchor(anchor, scope, baseUrl, 'episode');
            if (!item) continue;
            const row = auxiliaryRowForAnchor(anchor, scope);
            const rowText = item.rawText || item.title;
            const episodeNumber = rowText.match(/\bS\s*\d+\s*[,·.]?\s*E(?:p)?\s*\d+\b/i)
                || rowText.match(/\b\d+\.\d+\b/);
            const href = anchor.getAttribute('href') || anchor.href || '';
            const explicitEpisodeContext = !!anchor.closest?.(
                '[data-testid*="episode" i], [class*="episode" i], .eplist, #episodes_content'
            );
            const explicitEpisodeReference = /[?&]ref_=ttep_ep\d+/i.test(href);

            // An episodes page can also contain recommendations and other
            // title links. Only accept an item when its own row carries an
            // episode number, its link has IMDb's episode ref, or it belongs
            // to a clearly episode-specific container. This deliberately
            // prefers a missed card over polluting the filmography with recs.
            if (!episodeNumber && !explicitEpisodeContext && !explicitEpisodeReference) continue;

            item.episode = episodeNumber?.[0] || '';
            item.airDate = (rowText.match(/\b(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*\s+\d{1,2},?\s+\d{4}\b/i) || [])[0] || '';
            const key = item.href.toLowerCase();
            if (seen.has(key)) continue;
            seen.add(key);
            items.push(item);
        }
        return items;
    }

    function extractAuxiliaryQuotes(doc, baseUrl) {
        const scope = doc.querySelector('main') || doc.querySelector('#main') || doc.body || doc;
        const candidates = Array.from(scope.querySelectorAll('blockquote, [data-testid*="quote"], .sodatext, .quote'));
        // The older quote view did not mark entries semantically. Its list
        // items are a fallback, but short controls/navigation are rejected.
        if (!candidates.length) {
            candidates.push(...Array.from(scope.querySelectorAll('li, tr')).filter(node => auxiliaryNodeText(node).length >= 45));
        }

        const quotes = [];
        const seen = new Set();
        for (const candidate of candidates) {
            const nestedQuote = candidate.querySelector?.('blockquote, .sodatext, .quote, [data-testid*="quote"]');
            if (nestedQuote && nestedQuote !== candidate) continue;
            const quoteText = auxiliaryNodeText(candidate);
            if (quoteText.length < 18 || quoteText.length > 5000) continue;
            if (/^(?:quotes?|add a quote|edit|report|see all)$/i.test(quoteText)) continue;
            const context = candidate.closest?.('li, article, [data-testid*="quote"], .sodatext, .quote') || candidate;
            const links = Array.from(context.querySelectorAll?.('a[href]') || []);
            const speaker = links.find(a => /\/name\/nm\d+/i.test(a.getAttribute('href') || a.href || ''));
            const source = links.find(a => /\/title\/tt\d+/i.test(a.getAttribute('href') || a.href || ''));
            const speakerHref = speaker ? auxiliarySafeLocalHref(speaker, baseUrl) : '';
            const sourceHref = source ? auxiliarySafeLocalHref(source, baseUrl) : '';
            const speakerName = speaker ? auxiliaryNodeText(speaker) : '';
            const sourceTitle = source ? auxiliaryNodeText(source) : '';
            const key = quoteText.toLowerCase();
            if (seen.has(key)) continue;
            seen.add(key);
            quotes.push({
                text: quoteText,
                speaker: speakerName,
                speakerHref,
                source: sourceTitle,
                sourceHref,
                context: auxiliaryNodeText(context) !== quoteText ? auxiliaryNodeText(context) : ''
            });
        }
        return quotes;
    }

    function parseAuxiliaryPageFromDom(doc, sourceUrl = location.href, forcedKind = '') {
        const url = new URL(sourceUrl, location.href);
        const kind = forcedKind || classifyImdbPath(url.pathname);
        const titleId = getTitleIdFromUrl(url.href);
        const nameId = getNameIdFromUrl(url.href);
        const mainHeading = auxiliaryNodeText(doc.querySelector('main h1, #main h1, h1'));
        const documentTitle = normalizeAuxiliaryText(doc.title || '');
        const cleanDocumentTitle = documentTitle
            .replace(/\s*[-|–]\s*(?:Full Cast & Crew|Full Credits|Episodes|Quotes)(?:\s*[-|–]\s*IMDb)?\s*$/i, '')
            .replace(/\s+-\s+IMDb\s*$/i, '')
            .trim();
        const ogTitle = normalizeAuxiliaryText(doc.querySelector('meta[property="og:title"]')?.getAttribute('content') || '');
        const identity = mainHeading && !/^(?:full cast(?:\s*&\s*crew)?|cast\s*&\s*crew|full credits|episodes|quotes|quotes for .+)$/i.test(mainHeading)
            ? mainHeading
            : cleanDocumentTitle || ogTitle || mainHeading;
        const titlePage = kind.startsWith('title-');
        const model = {
            type: 'auxiliary',
            pageKind: kind,
            id: titlePage ? titleId : nameId,
            title: titlePage ? identity : '',
            name: !titlePage ? identity : '',
            heading: kind === 'title-fullcredits' ? 'Full Cast & Crew'
                : kind === 'name-fullcredits' ? 'Full Credits'
                : kind === 'title-episodes' ? 'Episodes'
                : 'Quotes',
            subheading: identity,
            year: (documentTitle.match(/\((\d{4})(?:[–-]\d{4})?\)/) || [])[1] || '',
            sections: [],
            items: [],
            quotes: [],
            season: url.searchParams.get('season') || '',
            page: url.searchParams.get('page') || '',
            seasonLinks: [],
            nextHref: '',
            previousHref: '',
            notice: ''
        };

        if (kind === 'title-fullcredits' || kind === 'name-fullcredits') {
            model.sections = extractAuxiliarySections(doc, url.href, kind);
        } else if (kind === 'title-episodes') {
            model.items = extractAuxiliaryEpisodes(doc, url.href, titleId);
            const seasonAnchors = Array.from((doc.querySelector('main') || doc.body || doc).querySelectorAll('a[href]'));
            const seasonMap = new Map();
            for (const anchor of seasonAnchors) {
                const href = auxiliarySafeLocalHref(anchor, url.href);
                const parsedHref = (() => { try { return new URL(anchor.href, url.href); } catch (err) { return null; } })();
                if (!href || !parsedHref || !/\/episodes\/?$/i.test(parsedHref.pathname)) continue;
                const season = parsedHref.searchParams.get('season');
                if (season !== null) seasonMap.set(season, {
                    label: auxiliaryNodeText(anchor) || (season === 'Unknown' ? 'Unknown season' : `Season ${season}`),
                    href,
                    current: season === (url.searchParams.get('season') || '')
                });
            }
            model.seasonLinks = Array.from(seasonMap.values());
            const next = doc.querySelector('a[rel="next"], a[data-testid*="next"]');
            const previous = doc.querySelector('a[rel="prev"], a[rel="previous"], a[data-testid*="previous"]');
            model.nextHref = next ? (() => { try { const u = new URL(next.href, url.href); return u.origin === url.origin ? `${u.pathname}${u.search}` : ''; } catch (err) { return ''; } })() : '';
            model.previousHref = previous ? (() => { try { const u = new URL(previous.href, url.href); return u.origin === url.origin ? `${u.pathname}${u.search}` : ''; } catch (err) { return ''; } })() : '';
        } else if (kind === 'title-quotes' || kind === 'name-quotes') {
            model.quotes = extractAuxiliaryQuotes(doc, url.href);
        }

        const scopeText = auxiliaryNodeText(doc.querySelector('main') || doc.body || doc);
        if (!model.items.length && !model.sections.length && !model.quotes.length) {
            const emptyText = scopeText.match(/(?:no|don't have|do not have|has no)\s+(?:any\s+)?(?:quotes|episodes|credits|cast|crew)[^.!?]*/i)?.[0];
            if (emptyText) model.notice = emptyText;
        }
        return model;
    }

    function hasAuxiliaryPageContent(model) {
        return !!model && (
            safeArray(model.items).length > 0 ||
            safeArray(model.sections).some(section => safeArray(section?.items).length > 0) ||
            safeArray(model.quotes).length > 0 ||
            !!model.notice
        );
    }

    function auxiliaryPageNavigation(model) {
        const titleId = model?.pageKind?.startsWith('title-') ? model.id : '';
        const nameId = model?.pageKind?.startsWith('name-') ? model.id : '';
        const links = [];
        if (titleId) {
            links.push({ label: 'Title overview', href: `/title/${titleId}/` });
            links.push({ label: 'Full credits', href: `/title/${titleId}/fullcredits/` });
            links.push({ label: 'Episodes', href: `/title/${titleId}/episodes/` });
            links.push({ label: 'Quotes', href: `/title/${titleId}/quotes/` });
        } else if (nameId) {
            links.push({ label: 'Person overview', href: `/name/${nameId}/` });
            links.push({ label: 'Full credits', href: `/name/${nameId}/fullcredits/` });
            links.push({ label: 'Quotes', href: `/name/${nameId}/quotes/` });
        }
        return links.map(link => `<a class="tm-aux-nav-link${link.href === `${model.pageKind?.startsWith('title-') ? '/title/' + model.id : '/name/' + model.id}/${model.pageKind?.slice(model.pageKind.indexOf('-') + 1)}/` ? ' is-current' : ''}" href="${htmlEscape(link.href)}">${htmlEscape(link.label)}</a>`).join('');
    }

    function renderAuxiliaryItem(item, kind) {
        const href = String(item?.href || '#');
        const label = text(item?.title || item?.name || 'Untitled');
        const note = text(item?.note || item?.subtitle || '');
        const image = item?.image ? `<img class="tm-aux-item-image" ${imgAttr(item.image, 76, label)}>` : '';
        const details = [item?.episode, item?.airDate, item?.year].filter(Boolean).filter((v, i, a) => a.indexOf(v) === i).join(' · ');
        return `
            <li class="tm-aux-item">
                ${image}
                <div class="tm-aux-item-copy">
                    <a class="tm-aux-item-title" href="${htmlEscape(href)}">${htmlEscape(label)}</a>
                    ${details ? `<div class="tm-aux-item-meta">${htmlEscape(details)}</div>` : ''}
                    ${note ? `<div class="tm-aux-item-note">${htmlEscape(note)}</div>` : ''}
                </div>
            </li>
        `;
    }

    function renderAuxiliaryPage(model) {
        const nav = auxiliaryPageNavigation(model);
        const identity = text(model?.subheading || model?.title || model?.name);
        const heading = text(model?.heading || 'IMDb');
        let content = '';

        if (model.pageKind === 'title-quotes' || model.pageKind === 'name-quotes') {
            content = safeArray(model.quotes).length
                ? `<article class="tm-card tm-aux-quotes"><h2>Quotes (${model.quotes.length})</h2><ol class="tm-aux-quote-list">${model.quotes.map(quote => `
                    <li class="tm-aux-quote">
                        <blockquote>${htmlEscape(quote.text)}</blockquote>
                        <div class="tm-aux-quote-attribution">
                            ${quote.speakerHref ? `<a href="${htmlEscape(quote.speakerHref)}">${htmlEscape(quote.speaker)}</a>` : htmlEscape(quote.speaker || '')}
                            ${quote.sourceHref ? ` <span>—</span> <a href="${htmlEscape(quote.sourceHref)}">${htmlEscape(quote.source)}</a>` : quote.source ? ` <span>— ${htmlEscape(quote.source)}</span>` : ''}
                        </div>
                    </li>`).join('')}</ol></article>`
                : '';
        } else if (model.pageKind === 'title-episodes') {
            const seasonNav = safeArray(model.seasonLinks).length
                ? `<nav class="tm-aux-season-nav" aria-label="Episode seasons">${model.seasonLinks.map(link => `<a href="${htmlEscape(link.href)}" class="${link.current ? 'is-current' : ''}">${htmlEscape(link.label)}</a>`).join('')}</nav>`
                : '';
            const items = safeArray(model.items);
            content = `${seasonNav}<article class="tm-card tm-aux-list"><h2>Episodes (${items.length})${model.season ? ` · Season ${htmlEscape(model.season)}` : ''}</h2>${items.length ? `<ol class="tm-aux-item-list">${items.map(item => renderAuxiliaryItem(item, 'episode')).join('')}</ol>` : ''}${model.previousHref || model.nextHref ? `<div class="tm-aux-pagination">${model.previousHref ? `<a href="${htmlEscape(model.previousHref)}">← Previous</a>` : ''}${model.nextHref ? `<a href="${htmlEscape(model.nextHref)}">Next →</a>` : ''}</div>` : ''}</article>`;
        } else {
            content = safeArray(model.sections).map(section => `
                <article class="tm-card tm-aux-list">
                    <h2>${htmlEscape(section.heading)} <span class="tm-aux-count">(${safeArray(section.items).length})</span></h2>
                    <ol class="tm-aux-item-list">${safeArray(section.items).map(item => renderAuxiliaryItem(item, model.pageKind)).join('')}</ol>
                </article>
            `).join('');
        }

        return `
            <div class="tm-shell tm-auxiliary-page">
                ${renderTopbar(text($(SEARCH_INPUT_SELECTOR)?.value))}
                <nav class="tm-aux-nav" aria-label="IMDb sections">${nav}</nav>
                <section class="tm-hero tm-aux-hero">
                    <div class="tm-hero-title">
                        <div class="tm-kicker">IMDb ${htmlEscape(model.pageKind.replace('-', ' · '))}</div>
                        <h1 class="tm-title">${htmlEscape(heading)}</h1>
                        ${identity ? `<div class="tm-aux-subtitle">${htmlEscape(identity)}</div>` : ''}
                    </div>
                </section>
                ${model.notice ? `<article class="tm-card tm-aux-empty"><h2>No entries available</h2><p>${htmlEscape(model.notice)}</p></article>` : ''}
                ${content || (!model.notice ? `<article class="tm-card tm-aux-empty"><h2>No entries could be extracted</h2><p>IMDb's page structure may have changed, or this section has no entries in the currently loaded document. Open the original IMDb page to inspect it.</p></article>` : '')}
                <div class="tm-aux-footer"><a href="${htmlEscape(model.pageKind.startsWith('title-') ? `/title/${model.id}/` : `/name/${model.id}/`)}">← Back to IMDb overview</a></div>
            </div>
        `;
    }

    function renderLinkedPeople(items) {
        const list =
            safeArray(items)
                .filter(item => item?.text);

        if (!list.length) {
            return '<span class="tm-muted">Not available</span>';
        }

        return list
            .map(item =>
                item.href
                    ? `<a href="${htmlEscape(item.href)}">${htmlEscape(item.text)}</a>`
                    : htmlEscape(item.text)
            )
            .join(', ');
    }

    function renderTileGrid(
        items,
        options = {}
    ) {
        const list =
            safeArray(items)
                .filter(
                    item =>
                        item &&
                        (
                            item.title ||
                            item.text
                        )
                );

        if (!list.length) {
            return '<p class="tm-muted">Not available</p>';
        }

        const gridClass =
            options.gridClass ||
            'tm-related-films-grid';

        const tileClass =
            options.tileClass ||
            '';

        const titleKey =
            options.titleKey ||
            'title';

        const titleBuilder =
            typeof options.titleBuilder === 'function'
                ? options.titleBuilder
                : item =>
                    text(
                        item[titleKey] ||
                        item.text
                    );

        const metaBuilder =
            typeof options.metaBuilder === 'function'
                ? options.metaBuilder
                : () => '';

        const showTitle =
            options.showTitle !== false;

        const showMeta =
            options.showMeta !== false;

        const imageWidth =
            Number(
                options.imageWidth
            ) || 240;

        return `
            <div class="${htmlEscape(gridClass)}">
                ${list.map(item => {
                    const href =
                        normalizeHref(
                            item.href || ''
                        );

                    const tag =
                        href ? 'a' : 'div';

                    const label =
                        titleBuilder(item);

                    let imageAttrOrFallback =
                        '';

                    if (item.image) {
                        imageAttrOrFallback =
                            `<img ${imgAttr(item.image, imageWidth, label)}>`;
                    } else if (
                        options.fallbackImage
                    ) {
                        imageAttrOrFallback =
                            `<img src="${DEFAULT_AVATAR}" alt="No image available" style="object-fit:contain; background:#252525;">`;
                    } else if (
                        options.fallbackPoster
                    ) {
                        imageAttrOrFallback =
                            `<div class="tm-tile-placeholder" aria-label="No poster available">${DEFAULT_POSTER_SVG}</div>`;
                    }

                    const meta =
                        text(
                            metaBuilder(item)
                        );

                    const bodyMarkup =
                        showTitle ||
                        (
                            showMeta &&
                            meta
                        )
                            ? `
                                <span class="tm-tile-body">
                                    ${
                                        showTitle
                                            ? `<span class="tm-tile-title">${htmlEscape(label)}</span>`
                                            : ''
                                    }

                                    ${
                                        showMeta && meta
                                            ? `<span class="tm-tile-meta">${meta}</span>`
                                            : ''
                                    }
                                </span>
                            `
                            : '';

                    return `
                        <${tag}${href ? ` href="${htmlEscape(href)}"` : ''} class="tm-tile${tileClass ? ' ' + tileClass : ''}">
                            ${imageAttrOrFallback}
                            ${bodyMarkup}
                        </${tag}>
                    `;
                }).join('')}
            </div>
        `;
    }

    function hasReviewContent(model) {
        return (
            !!text(model?.summary || '') ||
            !!text(model?.summaryBody || '')
        );
    }

    function hasSynopsisContent(model) {
        const synopsis =
            normalizeSpace(
                model?.synopsis || ''
            );

        const plot =
            normalizeSpace(
                model?.plot || ''
            );

        if (
            !synopsis ||
            synopsis.length < 40
        ) {
            return false;
        }

        if (!plot) {
            return true;
        }

        if (sameText(synopsis, plot)) {
            return false;
        }

        const synopsisLower =
            synopsis.toLowerCase();

        const plotLower =
            plot.toLowerCase();

        if (
            synopsisLower ===
            plotLower
        ) {
            return false;
        }

        if (
            synopsisLower.includes(
                plotLower
            ) &&
            synopsis.length <=
                plot.length + 20
        ) {
            return false;
        }

        return true;
    }

    function renderTopbar(searchValue) {
        const timeTaken =
            typeof loadStartTimeMs !== 'undefined'
                ? (
                    (
                        Date.now() -
                        loadStartTimeMs
                    ) /
                    1000
                ).toFixed(2)
                : '0.00';

        const preserveScroll =
            getSetting(
                'preserveScroll',
                PRESERVE_SCROLL_BACK_FORWARD
            );

        const hideContent =
            getSetting(
                'hideContent',
                HIDE_CONTENT_ON_NAVIGATE
            );

        void preserveScroll;
        void hideContent;

        return `
            <div class="tm-topbar">
                <div style="display:flex; align-items:center; gap:12px;">
                    <div class="tm-brand">IMDb</div>
                    <div class="tm-timer">Loaded in ${timeTaken}s</div>

    }

            document.getElementById(

                    document.getElementById(
                        ROOT_ID
                    );

                if (!retryRoot) {
                    debugLog(
                        'Root still missing after body became available.'
                    );

                    return;
                }

                startInitialRouteHandling();

            return;

        const navType =
            getNavigationType();

        const shouldRestoreScroll =
            navType === 'back_forward' &&
            PRESERVE_SCROLL_BACK_FORWARD;

            scrollBehavior:
                shouldRestoreScroll

                    : 'reset',

            preferCache:
                navType ===

    function destroyIframe(frame) {
        if (!frame) {
            return;
        }

        try {
            frame.contentWindow?.stop?.();
        } catch (err) {}

        try {
            frame.srcdoc = '';
        } catch (err) {}

        try {
            frame.src = 'about:blank';
        } catch (err) {}

        try {
            const doc =
                frame.contentDocument;

            if (doc) {
                doc.open();
                doc.write('');
                doc.close();
            }
        } catch (err) {}

        try {
            frame.remove();
        } catch (err) {}
    }

    function isSessionCachePayloadUsable(
        payload
    ) {
        if (
            !payload ||
            !payload.model
        ) {
            return false;
        }

        if (
            Date.now() -
                (payload.savedAt || 0) >
            SESSION_CACHE_TTL_MS
        ) {
            return false;
        }

        if (!pageKind || payload.pageType !== pageKind) {
            return false;
        }

        if (pageKind === 'title') return !!payload.model.title;
        if (pageKind === 'name') return !!payload.model.name;
        if (isAuxiliaryPage) return hasAuxiliaryPageContent(payload.model);
        return false;
    }

            return;
        }

        root.innerHTML = isAuxiliaryPage
            ? renderAuxiliaryPage(model)
            : isTitlePage
                ? renderTitlePage(model)
                : renderNamePage(model);

        root.style.display =
            'block';

        applyPostRenderEnhancements(
            root
        );

        // The replacement is now in the DOM and visible; allow normal page
        // painting again. The native IMDb app remains hidden by the main CSS.
        releaseEarlyPageShield();

        if (persistCache) {
            persistPageCache(model);
        }

        if (pageKind === 'title') {
            renderTrailer();

            if (
                !trailerVideoId &&
                EMBED_YOUTUBE_TRAILER
            ) {
                startTrailerSearch(
                    model.title,
                    model.year
                );
            }
        }

        const shouldRestoreScroll =
            scrollBehavior ===
                'restore' &&
            getSetting(
                'preserveScroll',
                PRESERVE_SCROLL_BACK_FORWARD
            );

        if (shouldRestoreScroll) {
            const scrollY =
                loadScrollPosition();

            if (scrollY) {
                requestAnimationFrame(
                    () =>
                        window.scrollTo(
                            0,
                            scrollY
                        )
                );
            }
        } else {
            window.scrollTo(0, 0);
        }
    }

    function renderFromSessionCache(
        payload,
        options = {}
    ) {
        resetTrailerState();

        trailerVideoId =
            payload?.trailerVideoId ||
            null;

        trailerRequestStarted = false;

        renderModel(
            payload?.model,
            options
        );
    }

    function startExtractionForCurrentRoute() {
        refreshRouteState();

        resetTrailerState();

        teardownExtraction();

        const root =
            document.getElementById(
                ROOT_ID
            );

        const hideContent =
            getSetting(
                'hideContent',
                HIDE_CONTENT_ON_NAVIGATE
            );

        if (!root) {
            debugLog(
                'startExtractionForCurrentRoute: root missing.'
            );

            return;
        }

        if (hideContent) {
            root.style.display =
                'none';

            root.innerHTML = '';
        } else {
            root.innerHTML =
                renderLoading();

            root.style.display =
                'block';
        }

        if (!hideContent) {
            window.scrollTo(
                0,
                0
            );
        }

        if (
            !EXPERIMENTAL_NO_IFRAME_EXTRACTION
        ) {
            debugLog(
                'Initializing extraction for:',
                path
            );

            iframe =
                document.createElement(
                    'iframe'
                );

            iframe.style.cssText =
                'position: fixed; width: 1px; height: 1px; opacity: 0; pointer-events: none; top: 0; left: 0; z-index: -9999; border: 0;';

            iframe.src =
                location.href;

            document.body.appendChild(
                iframe
            );
        }

        if (
            pageKind === 'title' &&
            !DISABLE_SYNOPSIS_IFRAME
        ) {
            const tconstMatch =
                path.match(
                    /\/title\/(tt\d+)/
                );

            if (tconstMatch) {
                fetch(
                    `/title/${tconstMatch[1]}/plotsummary`
                )
                    .then(r =>
                        r.text()
                    )
                    .then(html => {
                        fetchedSynopsisHtml =
                            html;
                    })
                    .catch(err =>
                        debugLog(
                            'Synopsis fetch failed',
                            err
                        )
                    );
            }
        }

        extractionStartedAt =
            Date.now();

        loadStartTimeMs =
            Date.now();

        startTitleLoading();
        startExtractionPolling();
    }

    function handleRouteChange(
        options = {}
    ) {
        const scrollBehavior =
            options.scrollBehavior ||
            'reset';

        const preferCache =
            !!options.preferCache;

        refreshRouteState();

        if (!pageKind) {
            return;
        }

        const payload =
            readSessionJson(
                cacheKeyForUrl()
            );

        if (
            preferCache &&
            isSessionCachePayloadUsable(
                payload
            )
        ) {
            return renderFromSessionCache(
                payload,
                {
                    scrollBehavior
                }
            );
        }

        // FIX (v1.6, Firefox/SPA):
        //
        // When this route change comes from an in-page SPA navigation
        // (pushState / popstate) there has been NO document load. The
        // live document still belongs to the PREVIOUS route, so
        // startExtractionForCurrentRoute() would poll the OLD page's
        // #__NEXT_DATA__ and render stale data (or hang on loading).
        //
        // With EXPERIMENTAL_NO_IFRAME_EXTRACTION enabled there is no
        // hidden iframe either, so the only correct option is to fetch
        // the destination HTML and parse it off-thread.
        if (options.spaNavigation) {
            return navigateViaFetch(
                new URL(location.href),
                {
                    replace: true,
                    scrollBehavior
                }
            );
        }

        startExtractionForCurrentRoute();
    }

    // ---------------------------------------------------------------------
    // FIX (v1.6): SPA navigation with a real data source.
    //
    // Fetches the destination page HTML (same-origin, cookies included),
    // parses it with DOMParser and extracts the model with the exact same
    // parsers used by the fallback paths. Works identically in Firefox and
    // Chrome because it never depends on a stale document.
    //
    // If anything fails, it falls back to a full page load, which is the
    // well-tested document-start path on every browser.
    // ---------------------------------------------------------------------

    let navigateFetchToken = 0;

    async function navigateViaFetch(
        dest,
        options = {}
    ) {
        const token =
            ++navigateFetchToken;

        const replace =
            !!options.replace;

        const scrollBehavior =
            options.scrollBehavior ||
            'reset';

        const nextUrl =
            `${dest.pathname}${dest.search || ''}`;

        persistScrollPosition();

        resetTrailerState();

        teardownExtraction();

        startTitleLoading();

        try {
            const res =
                await fetch(
                    nextUrl,
                    {
                        credentials:
                            'same-origin',
                        redirect:
                            'follow'
                    }
                );

            if (!res.ok) {
                throw new Error(
                    `HTTP ${res.status}`
                );
            }

            const html =
                await res.text();

            // A newer navigation superseded this one.
            if (token !== navigateFetchToken) {
                return;
            }

            const nextPageKind = classifyImdbPath(dest.pathname);
            const nextIsTitle = nextPageKind === 'title';
            const nextIsName = nextPageKind === 'name';
            const nextIsAuxiliary = !!nextPageKind && !nextIsTitle && !nextIsName;

            if (!nextPageKind) {
                throw new Error('Unsupported destination');
            }

            const doc =
                new DOMParser().parseFromString(
                    html,
                    'text/html'
                );

            const nextData = tryJSON(
                doc.querySelector(NEXT_DATA_SELECTOR)?.textContent || ''
            );

            let parsed;
            if (nextIsAuxiliary) {
                parsed = parseAuxiliaryPageFromDom(doc, dest.href, nextPageKind);
            } else if (nextIsTitle) {
                parsed = enhanceTitleModelFromDom(parseTitleDataFromDom(doc), doc);
            } else {
                const nameModel = nextData
                    ? parseNameData(nextData, dest.href)
                    : parseNameDataFromDom(doc, dest.href);
                parsed = enhanceNameModelFromDom(nameModel, doc, dest.href);
            }

            const hasIdentity = nextIsAuxiliary
                ? hasAuxiliaryPageContent(parsed)
                : nextIsTitle
                    ? !!parsed.title
                    : !!parsed.name;

            if (!hasIdentity) {
                throw new Error(
                    'Extraction returned no identity'
                );
            }

            refreshRouteState(nextUrl);

            if (replace) {
                history.replaceState(
                    {
                        tmImdbClean: true
                    },
                    '',
                    nextUrl
                );
            } else {
                history.pushState(
                    {
                        tmImdbClean: true
                    },
                    '',
                    nextUrl
                );
            }

            stopTitleLoading();

            renderModel(parsed, {
                scrollBehavior,
                persistCache: true
            });

            // Preserve an in-progress search query across SPA
            // navigation, mirroring the old finalizeWithParsed().
            const searchValue =
                document.querySelector(
                    '#suggestion-search'
                )?.value || '';

            if (searchValue) {
                const renderedInput =
                    document
                        .getElementById(ROOT_ID)
                        ?.querySelector(
                            'input[name="q"]'
                        );

                if (renderedInput) {
                    renderedInput.value =
                        searchValue;
                }
            }
        } catch (err) {
            debugLog(
                'navigateViaFetch failed, falling back to full page load:',
                err
            );

            if (token === navigateFetchToken) {
                location.assign(dest.href);
            }
        }
    }

    function navigateTo(
        url,
        options = {}
    ) {
        const replace =
            !!options.replace;

        const dest =
            url instanceof URL
                ? url
                : new URL(
                    String(
                        url || ''
                    ),
                    location.href
                );

        if (
            dest.origin !==
            location.origin
        ) {
            return;
        }

        if (!isSupportedPagePath(dest.pathname)) {
            return;
        }


        if (
            dest.hash &&
            (
                dest.hash.includes(
                    'noredirects'
                ) ||
                dest.hash.includes(
                    'noredicts'
                )
            )
        ) {
            return;
        }

        const nextUrl =
            `${dest.pathname}${dest.search || ''}`;

        // FIX (v1.6):
        //
        // The old code pushed the new URL into history and then ran
        // handleRouteChange(), which polled the CURRENT document. With
        // EXPERIMENTAL_NO_IFRAME_EXTRACTION that document is still the
        // PREVIOUS page, so cache misses produced stale/blank renders
        // (most visible on Firefox, where the session cache was cold).
        //
        // New behaviour: use the session cache when it actually contains
        // the destination; otherwise fetch + parse the destination and
        // render it for real.
        refreshRouteState(nextUrl);

        const payload =
            readSessionJson(
                cacheKeyForUrl(nextUrl)
            );

        if (
            isSessionCachePayloadUsable(
                payload
            )
        ) {
            persistScrollPosition();

            resetTrailerState();

            teardownExtraction();

            if (replace) {
                history.replaceState(
                    {
                        tmImdbClean: true
                    },
                    '',
                    nextUrl
                );
            } else {
                history.pushState(
                    {
                        tmImdbClean: true
                    },
                    '',
                    nextUrl
                );
            }

            return renderFromSessionCache(
                payload,
                {
                    scrollBehavior:
                        'reset'
                }
            );
        }

        navigateViaFetch(dest, {
            replace,
            scrollBehavior:
                'reset'
        });
    }

    function shouldInterceptLinkClick(
        e,
        anchor
    ) {
        if (!anchor) {
            return false;
        }

        if (e.defaultPrevented) {
            return false;
        }

        if (e.button !== 0) {
            return false;
        }

        if (
            e.metaKey ||
            e.ctrlKey ||
            e.shiftKey ||
            e.altKey
        ) {
            return false;
        }

        if (
            anchor.hasAttribute(
                'download'
            )
        ) {
            return false;
        }

        const target =
            anchor.getAttribute(
                'target'
            );

        if (
            target &&
            target !== '_self'
        ) {
            return false;
        }

        return true;
    }

            });

    document.addEventListener(
        'click',
        e => {
            const root =
                document.getElementById(
                    ROOT_ID
                );

            const anchor =
                e.target?.closest?.(
                    'a[href]'
                );

            if (
                !root ||
                !anchor ||
                !root.contains(anchor)
            ) {
                return;
            }

            if (
                !shouldInterceptLinkClick(
                    e,
                    anchor
                )
            ) {
                return;
            }

            let dest;

            try {
                dest =
                    new URL(
                        anchor.getAttribute(
                            'href'
                        ),
                        location.href
                    );
            } catch (err) {
                return;
            }

            if (
                dest.origin !==
                location.origin
            ) {
                return;
            }
            if (!isSupportedPagePath(dest.pathname)) {
                return;
            }


            if (
                dest.hash &&
                (
                    dest.hash.includes(
                        'noredirects'
                    ) ||
                    dest.hash.includes(
                        'noredicts'
                    )
                )
            ) {
                return;
            }

            e.preventDefault();

            navigateTo(dest);
        },
        true
    );

    function startTrailerSearch(
        title,
        year
    ) {
        if (
            trailerRequestStarted ||
            pageKind !== 'title' ||
            !EMBED_YOUTUBE_TRAILER ||
            typeof GM_xmlhttpRequest ===
                'undefined'
        ) {
            return;
        }

        if (!title) {
            return;
        }

        trailerRequestStarted =
            true;

        debugLog(
            'Starting parallel trailer search for:',
            title
        );

        const query =
            encodeURIComponent(
                `"${title}"${
                    year
                        ? ` (${year})`
                        : ''
                } movie trailer site:youtube.com`
            );

        trailerRequest =
            GM_xmlhttpRequest({
                method: "GET",

                url:
                    `https://www.google.com/search?btnI=1&q=${query}`,

                headers: {
                    "User-Agent":
                        navigator.userAgent
                },

                onload: function (res) {
                    let match =
                        (
                            res.finalUrl ||
                            ''
                        ).match(
                            /v=([a-zA-Z0-9_-]{11})/
                        );

                    if (!match) {
                        match =
                            res.responseText
                                .match(
                                    /youtube\.com\/watch\?v=([a-zA-Z0-9_-]{11})/
                                );
                    }

                    if (match) {
                        trailerVideoId =
                            match[1];

                        renderTrailer();

                        updateCachedTrailer(
                            trailerVideoId
                        );
                    }
                }
            });
    }

    function renderTrailer() {
        const trailerDiv =
            document.getElementById(
                'tm-dynamic-trailer'
            );

        if (
            trailerDiv &&
            trailerVideoId
        ) {
            trailerDiv.style.display =
                'block';

            trailerDiv.innerHTML = `
                <iframe
                    width="100%"
                    height="100%"
                    src="https://www.youtube.com/embed/${trailerVideoId}?autoplay=0"
                    frameborder="0"
                    allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
                    allowfullscreen
                ></iframe>
            `;

            const hero =
                trailerDiv.closest(
                    '.tm-hero'
                );

            if (hero) {
                hero.classList.add(
                    'tm-has-trailer'
                );
            }
        }
    }

    function extractionElapsedMs() {
        return (
            Date.now() -
            extractionStartedAt
        );
    }

    function hasTimedOut() {
        return (
            extractionElapsedMs() >
            MAX_EXTRACTION_WAIT_MS
        );
    }

    // ---------------------------------------------------------------------
    // Extraction
    // ---------------------------------------------------------------------

    function startExtractionPolling() {
        if (
            checkInterval ||
            (
                !EXPERIMENTAL_NO_IFRAME_EXTRACTION &&
                !iframe
            )
        ) {
            return;
        }

        checkInterval =
            setInterval(() => {
                if (pageHidden) {
                    return;
                }

                if (
                    !EXPERIMENTAL_NO_IFRAME_EXTRACTION &&
                    !iframe
                ) {
                    return;
                }

                const timedOut =
                    hasTimedOut();

                const iDoc =
                    EXPERIMENTAL_NO_IFRAME_EXTRACTION
                        ? document
                        : iframe.contentDocument;

                function showOriginalPage() {
                    if (checkInterval) {
                        clearInterval(
                            checkInterval
                        );

                        checkInterval =
                            null;
                    }

                    stopTitleLoading();

                    try {
                        document.title =
                            (
                                iDoc &&
                                iDoc.title
                            )
                                ? iDoc.title
                                : originalTitle;
                    } catch (e) {
                        document.title =
                            originalTitle;
                    }

                    const root =
                        document.getElementById(
                            ROOT_ID
                        );

                    if (
                        EXPERIMENTAL_NO_IFRAME_EXTRACTION
                    ) {
                        releaseEarlyPageShield();

                        const preloadStyle = document.getElementById(PRELOAD_STYLE_ID);
                        if (preloadStyle) preloadStyle.remove();

                        if (
                            document.body
                        ) {
                            document.body.classList.remove(
                                'tm-imdb-clean-mode'
                            );
                        }

                        if (root) {
                            root.style.display =
                                'none';
                        }

                        const app =
                            document.querySelector(
                                APP_SELECTOR
                            );

                        if (app) {
                            app.style.display =
                                'block';
                        }

                        const style =
                            document.getElementById(
                                STYLE_ID
                            );

                        if (style) {
                            style.remove();
                        }
                    } else {
                        if (root) {
                            root.style.display =
                                'none';
                        }

                        if (iframe) {
                            iframe.style.cssText =
                                'position: fixed; width: 100vw; height: 100vh; opacity: 1; pointer-events: auto; top: 0; left: 0; z-index: 9999; border: 0;';
                        }
                    }

                    if (
                        summaryIframe
                    ) {
                        destroyIframe(
                            summaryIframe
                        );

                        summaryIframe =
                            null;
                    }
                }

                if (!iDoc) {
                    if (timedOut) {
                        showOriginalPage();
                    }

                    return;
                }

                function finalizeWithParsed(
                    parsed
                ) {
                    if (checkInterval) {
                        clearInterval(
                            checkInterval
                        );

                        checkInterval =
                            null;
                    }

                    stopTitleLoading();

                    const timerMs =
                        typeof loadStartTimeMs !== 'undefined'
                            ? Date.now() -
                              loadStartTimeMs
                            : 0;

                    console.log(
                        `⏱️ IMDb extraction and render completed in ${(timerMs / 1000).toFixed(2)} seconds`
                    );

                    let markup = isAuxiliaryPage
                        ? renderAuxiliaryPage(parsed)
                        : pageKind === 'title'
                            ? renderTitlePage(parsed)
                            : renderNamePage(parsed);

                    const searchValue =
                        iDoc.querySelector(
                            '#suggestion-search'
                        )?.value || '';

                    if (searchValue) {
                        markup =
                            markup.replace(
                                /name="q" placeholder=".*?" value=""/,
                                `name="q" placeholder="Search IMDb (via Google)" value="${htmlEscape(searchValue)}"`
                            );
                    }

                    const label = parsed.title || parsed.name || parsed.heading || 'IMDb';
                    document.title = `IMDB | ${label}${parsed.year ? ` (${parsed.year})` : ''}`;

                    const root =
                        document.getElementById(
                            ROOT_ID
                        );

                    // FIX:
                    // Firefox can never crash the script merely because
                    // the custom root disappeared.
                    if (!root) {
                        debugLog(
                            'finalizeWithParsed aborted: ROOT_ID does not exist.'
                        );

                        showOriginalPage();
                        return;
                    }

                    root.innerHTML =
                        markup;

                    root.style.display =
                        'block';

                    applyPostRenderEnhancements(
                        root
                    );

                    releaseEarlyPageShield();

                    lastRenderedModel =
                        parsed;

                    persistPageCache(
                        parsed
                    );

                    if (iframe) {
                        destroyIframe(
                            iframe
                        );

                        iframe = null;
                    }

                    if (pageKind === 'title') {
                        renderTrailer();
                    }

                    window.scrollTo(
                        0,
                        0
                    );
                }

                function fallbackToDom() {
                    debugLog(
                        'Falling back to direct DOM extraction (Native document)...'
                    );

                    stopTitleLoading();

                    let parsed;
                    if (isAuxiliaryPage) {
                        parsed = parseAuxiliaryPageFromDom(iDoc, location.href, pageKind);
                    } else if (pageKind === 'title') {
                        parsed = enhanceTitleModelFromDom(parseTitleDataFromDom(iDoc), iDoc);
                    } else {
                        parsed = enhanceNameModelFromDom(parseNameDataFromDom(iDoc, location.href), iDoc, location.href);
                    }

                    const hasIdentity = isAuxiliaryPage
                        ? hasAuxiliaryPageContent(parsed)
                        : pageKind === 'title'
                            ? !!parsed.title
                            : !!parsed.name;

                    if (!hasIdentity) {
                        return showOriginalPage();
                    }

                    finalizeWithParsed(
                        parsed
                    );
                }

                // Auxiliary IMDb sub-pages do not share the overview page's
                // __NEXT_DATA__ model. Read their rendered, route-specific DOM
                // directly; query parameters (e.g. season=2) remain in the URL.
                if (isAuxiliaryPage) {
                    const auxiliaryModel = parseAuxiliaryPageFromDom(iDoc, location.href, pageKind);
                    if (hasAuxiliaryPageContent(auxiliaryModel)) {
                        return finalizeWithParsed(auxiliaryModel);
                    }
                    if (timedOut) {
                        debugLog('No supported rows found on auxiliary page; restoring native IMDb page.');
                        return showOriginalPage();
                    }
                    return;
                }

                // ---------------------------------------------------------
                // Core JSON extraction
                // ---------------------------------------------------------

                const script =
                    iDoc.querySelector(
                        NEXT_DATA_SELECTOR
                    );

                if (!script) {
                    if (!timedOut) {
                        return;
                    }

                    debugLog(
                        'Timeout reached: Falling back to DOM extraction (JSON missing)'
                    );

                    return fallbackToDom();
                }

                // textContent is preferable here, especially because
                // Firefox may expose script contents differently while the
                // page is still being constructed.
                const data =
                    tryJSON(
                        script.textContent
                    );

                if (!data) {
                    debugLog(
                        'Warning: Failed to parse __NEXT_DATA__ JSON'
                    );

                    if (timedOut) {
                        return fallbackToDom();
                    }

                    return;
                }

                let parsed;

                if (isTitlePage) {
                    parsed =
                        enhanceTitleModelFromDom(
                            parseTitleData(data),
                            iDoc
                        );
                } else {
                    parsed =
                        enhanceNameModelFromDom(
                            parseNameData(data, location.href),
                            iDoc,
                            location.href
                        );
                }

                if (
                    isTitlePage &&
                    parsed.title
                ) {
                    startTrailerSearch(
                        parsed.title,
                        parsed.year
                    );
                }

                if (
                    !parsed.id &&
                    !isTitlePage
                ) {
                    debugLog(
                        'Waiting for name ID in JSON...'
                    );

                    return;
                }

                const hasIdentity =
                    isTitlePage
                        ? !!parsed.title
                        : !!parsed.name;

                if (!hasIdentity) {
                    // The JSON script can appear before the profile's identity
                    // fields finish hydrating. Do not expose the original page
                    // prematurely; allow the normal extraction window to elapse.
                    if (!timedOut) {
                        return;
                    }
                    return fallbackToDom();
                }

                if (isTitlePage) {

                    if (
                        !DISABLE_SYNOPSIS_IFRAME &&
                        fetchedSynopsisHtml
                    ) {
                        const parser =
                            new DOMParser();

                        const sDoc =
                            parser.parseFromString(
                                fetchedSynopsisHtml,
                                'text/html'
                            );

                        const sSynopsis =
                            extractSynopsisFromSummaryPage(
                                sDoc
                            );

                        if (sSynopsis) {
                            parsed.synopsis =
                                preferLongerMultilineText(
                                    parsed.synopsis,
                                    sSynopsis
                                );

                            fetchedSynopsisHtml =
                                null;
                        }
                    }

                    const waitingForDetailedSummary =
                        isTitlePage &&
                        !DISABLE_SYNOPSIS_IFRAME &&
                        !fetchedSynopsisHtml;

                    if (
                        waitingForDetailedSummary &&
                        !timedOut
                    ) {
                        if (
                            extractionElapsedMs() %
                                1000 <
                            50
                        ) {
                            debugLog(
                                'Waiting for synopsis DOM...'
                            );
                        }

                        return;
                    }

                    const waitingForSynopsis =
                        !timedOut &&
                        (
                            waitingForDetailedSummary ||
                            (
                                extractionElapsedMs() <
                                    MIN_TITLE_SETTLE_MS &&
                                !hasSynopsisContent(
                                    parsed
                                )
                            )
                        );

                    if (
                        waitingForSynopsis
                    ) {
                        return;
                    }
                }

                debugLog(
                    'Finalizing with model:',
                    parsed.title ||
                    parsed.name
                );

                finalizeWithParsed(
                    parsed
                );

            }, CHECK_INTERVAL_MS);
    }

    // ---------------------------------------------------------------------
    // bfcache
    // ---------------------------------------------------------------------

    window.addEventListener(
        'pagehide',
        e => {
            pageHidden = true;

            persistScrollPosition();

            const trailerDiv =
                document.getElementById(
                    'tm-dynamic-trailer'
                );

            if (trailerDiv) {
                trailerDiv.innerHTML =
                    '';

                trailerDiv.style.display =
                    'none';

                const hero =
                    trailerDiv.closest(
                        '.tm-hero'
                    );

                if (hero) {
                    hero.classList.remove(
                        'tm-has-trailer'
                    );
                }
            }

            try {
                trailerRequest?.abort?.();
            } catch (err) {}

            const showingOriginalPage =
                !!iframe &&
                iframe.style.pointerEvents ===
                    'auto' &&
                iframe.style.opacity ===
                    '1';

            if (
                !showingOriginalPage
            ) {
                if (checkInterval) {
                    clearInterval(
                        checkInterval
                    );

                    checkInterval =
                        null;
                }

                if (iframe) {
                    destroyIframe(
                        iframe
                    );

                    iframe = null;
                }

                if (summaryIframe) {
                    destroyIframe(
                        summaryIframe
                    );

                    summaryIframe =
                        null;
                }
            }
        }
    );

    window.addEventListener(
        'pageshow',
        e => {
            pageHidden = false;

            const icons =
                document.querySelectorAll(
                    'link[rel*="icon"]'
                );

            icons.forEach(link => {
                const href =
                    link.href;

                link.href = '';
                link.href = href;
            });

            if (
                e?.persisted &&
                pageKind === 'title' &&
                EMBED_YOUTUBE_TRAILER &&
                trailerVideoId
            ) {
                renderTrailer();
            }
        }
    );

    // ---------------------------------------------------------------------
    // Ctrl+S
    // ---------------------------------------------------------------------

    document.addEventListener(
        'keydown',
        e => {
            if (
                (e.ctrlKey || e.metaKey) &&
                e.key === 's'
            ) {
                e.preventDefault();

                const root =
                    document.getElementById(
                        ROOT_ID
                    );

                if (
                    !root ||
                    !root.innerHTML
                ) {
                    return;
                }

                const fullHtml = `
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta
        name="viewport"
        content="width=device-width, initial-scale=1.0"
    >

    <title>
        ${document.title}
    </title>

    <link
        id="tm-favicon"
        href="${FAVICON_URL}"
        rel="icon"
    >

    <style>
        ${getStyles()}
    </style>
</head>

<body class="tm-imdb-clean-mode">

    <div id="${ROOT_ID}">
        ${root.innerHTML}
    </div>

</body>
</html>`;

                const blob =
                    new Blob(
                        [fullHtml],
                        {
                            type:
                                'text/html'
                        }
                    );

                const url =
                    URL.createObjectURL(
                        blob
                    );

                const a =
                    document.createElement(
                        'a'
                    );

                a.href = url;

                const filename =
                    `imdb-${pageKind || 'page'}-${Date.now()}.html`;

                a.download =
                    filename;

                document.body.appendChild(
                    a
                );

                a.click();

                document.body.removeChild(
                    a
                );

                URL.revokeObjectURL(
                    url
                );
            }
        }
    );

})();
